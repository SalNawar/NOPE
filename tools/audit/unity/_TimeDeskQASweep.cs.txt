// QA sweep (kept as text in tools/audit/unity; copied into Assets/Editor as _TimeDeskQASweep.cs only to run it, never committed
// there). Run 7, Saleh found the demo "very buggy": a repeatable play that drives the real game through everything a player
// touches, screenshots each state at 1920x1080 and 1280x720 and asserts something at every step. Run it through the job
// runner like the audit play: a job file holding "_TimeDeskQASweep.Run". Outputs go to <out root>/qa/<yyyyMMdd_HHmmss>/
// (the out root as the audit play's: Library/AuditBaseline/out_root.txt, else Library/AuditBaseline): qa_report.txt (a PASS
// or FAIL line per assertion, NOTE lines for what the game does not have, "done" at the end), qa_warnings.txt (every
// warning and error logged), shots/<area>/<name>_1080.png and _720.png, shots.tsv (area, name, file) and a contact sheet per
// area, contact_<area>.png (tools/audit/qa_contact_sheet.py, run at the end with the python on PATH).
//
// The path, with a fixed seed (12345), from a clean slot (the player's save and preferences set aside and restored):
// - title: the Title (Continue hidden with no save), New Run, the pet adoption (a refused name, the cat's coats, then the dog
//   Biscuit in the charcoal coat, not the default: the preview and the coat row follow the choice);
// - office, day 1: the briefing and its Bureau memo, the AVAILABLE start, the desk with a traveller, the stability
//   monitor's size (the Helix River), the wheel opened and every one of its choices with the speech bubble, inspect mode
//   with a passport field linked to a rulebook row, the passport picked up, dropped and moved to the reading view, the
//   rulebook's tabs, the city's look-left pan and back (only the DESK tab shows there), every pull tab (PC, stamps,
//   city), the dater rack (the label lip under the daters); then the full day to the shift report, every traveller
//   decided with the desk machine: with the rack out the passport's visa page lies under the daters and the rack covers
//   none of its fields or its photo; a dater dragged over a passport field (its ink over the printed box), a dater
//   pressed where it hangs (it prints on the booklet's visa page), the papers handed back (the hand-back decides), DETAIN once on the desk's covered button (the
//   cover flipped up, the button pressed); the citation slip (shown on the screen, its lines readable, its CITED stamp
//   slamming on once printed, ACKNOWLEDGE putting it away); the cameras' idle sway off by default, on and off again;
// - home, night 1: the bills (each toggled; Heating must leave Electricity alone), the pet corner (pat, play; the pet drawn
//   in its charcoal coat), the House
//   (a card selected, bought when the wallet allows), the Night Slots machine (idle, a lever pull, the spin landing,
//   the can't-afford state), Sleep;
// - office, day 5 (RunManager.JumpToDay: the scanner and every PC app are in): two papers overlapped both ways, two papers'
//   fields linked in inspect mode, the scanner drop, the rulebook's SEALS tab; the PC: every app opened from its icon
//   (its window inside the screen, its title bar reachable, dragged by it, closed), the case board's panels (the record,
//   the rules check, the cross-check, the overlay), Arrange icons (the grid, no icon under the waiting strip), the
//   Orders tree, a mail opened, Settings' every row;
// - title again: Continue shown with the save, and it resumes the run.
// Everywhere a screen is shot the overlay's kit controls are checked: none off the screen, no two of them overlapping at
// rest; on the desk no paper is drawn over one lying above it.
// Assertions that fail on main are the known bugs other tracks own: the sweep lists them, it fixes nothing.

    private const int Seed = 12345;
    private const string ViewName = "TimeDeskQASweep";
    private const string StepKey = "QASweep.Step";
    private const string DirKey = "QASweep.Dir";
    private const string FailsKey = "QASweep.Fails";
    private const string PassesKey = "QASweep.Passes";
    private const string HadSaveKey = "QASweep.HadSave";
    private const string SceneKey = "QASweep.Scene";
    private const string SeedKey = "QASweep.SeedBefore";
    private const string PrefsKey = "QASweep.Prefs";
    private const string ViewWidthKey = "QASweep.ViewWidth";
    private const string ViewHeightKey = "QASweep.ViewHeight";
    private const int Playing = 1;
    private const int Finishing = 2;

    /// <summary>The per-player preferences the sweep starts without and restores: Settings' language and motion and the desktop's layout (DesktopPreferences' keys), each under its stored name (PlayerPrefKeys).</summary>
    private static IEnumerable<string> PrefNames() =>
        new[] { "TimeDesk.UiLanguage", "TimeDesk.ReducedMotion", "TimeDesk.MotionIntensity", "TimeDesk.CameraSway" }
            .Concat(typeof(DesktopPreferences).GetFields(BindingFlags.Static | BindingFlags.NonPublic | BindingFlags.Public)
                .Where(f => f.IsLiteral && f.FieldType == typeof(string)).Select(f => (string)f.GetRawConstantValue()).Where(v => v.StartsWith("TimeDesk.")))
            .Select(PlayerPrefKeys.For);

    private static readonly Stack<IEnumerator> Script = new Stack<IEnumerator>();
    private static string Dir => SessionState.GetString(DirKey, Path.Combine(Root, "qa", "unnamed"));
    private static string ReportPath => Path.Combine(Dir, "qa_report.txt");
    private static string WarningsPath => Path.Combine(Dir, "qa_warnings.txt");
    private static string ShotsList => Path.Combine(Dir, "shots.tsv");
    private static string SaveBackup => Path.Combine(SaveSystem.Folder, "nope_save_backup_qasweep.json");

    private static string Root
    {
        get
        {
            string lib = Path.Combine(Directory.GetCurrentDirectory(), "Library", "AuditBaseline");
            string file = Path.Combine(lib, "out_root.txt");
            return File.Exists(file) ? File.ReadAllText(file).Trim() : lib;
        }
    }

    static _TimeDeskQASweep()
    {
        if (SessionState.GetInt(StepKey, 0) == 0)
            return;
        EditorApplication.update += Tick;
        Application.logMessageReceived += OnLog;
    }

    /// <summary>The job: one sweep into a new folder.</summary>
    public static void Run()
    {
        SessionState.SetString(DirKey, Path.Combine(Root, "qa", DateTime.Now.ToString("yyyyMMdd_HHmmss")));
        Directory.CreateDirectory(Path.Combine(Dir, "shots"));
        File.WriteAllText(ReportPath, $"start {DateTime.Now:s} seed {Seed}\n");
        File.WriteAllText(WarningsPath, string.Empty);
        File.WriteAllText(ShotsList, "area\tname\tfile\n");

        int hadSave = 0;
        for (int i = 0; i < SaveSystem.Files.Count; i++)
        {
            if (!File.Exists(SaveSystem.Files[i]))
                continue;
            hadSave |= 1 << i;
            File.Copy(SaveSystem.Files[i], SaveBackup + i, true);
            File.Delete(SaveSystem.Files[i]);
        }
        SessionState.SetInt(HadSaveKey, hadSave);
        var prefs = new List<string>();
        foreach (string key in PrefNames())
        {
            string kind = !PlayerPrefs.HasKey(key) ? "absent" : key.EndsWith(PlayerPrefKeys.For("TimeDesk.MotionIntensity")) ? "float" : "string";
            string value = kind == "float" ? PlayerPrefs.GetFloat(key).ToString("R", System.Globalization.CultureInfo.InvariantCulture) : kind == "string" ? PlayerPrefs.GetString(key) : "";
            prefs.Add(string.Join("\u0001", key, kind, value));
            PlayerPrefs.DeleteKey(key);
        }
        SessionState.SetString(PrefsKey, string.Join("\u0002", prefs));
        SessionState.SetInt(SeedKey, Resources.Load<RunConfigSO>(RunManager.ConfigResourcePath).fixedRunSeed);
        PlayModeWindow.GetRenderingResolution(out uint width, out uint height);
        SessionState.SetInt(ViewWidthKey, (int)width);
        SessionState.SetInt(ViewHeightKey, (int)height);
        PlayModeWindow.SetCustomRenderingResolution(1920, 1080, ViewName);
        SessionState.SetInt(FailsKey, 0);
        SessionState.SetInt(PassesKey, 0);
        SessionState.SetInt(StepKey, Playing);
        SessionState.SetString(SceneKey, SceneManager.GetActiveScene().path);
        EditorSceneManager.OpenScene("Assets/Scenes/TitleScene.unity", OpenSceneMode.Single);
        EditorApplication.update += Tick;
        Application.logMessageReceived += OnLog;
        EditorApplication.EnterPlaymode();
    }

    private static void Tick()
    {
        int step = SessionState.GetInt(StepKey, 0);
        if (step == Finishing)
        {
            if (!EditorApplication.isPlaying)
                Finish();
            return;
        }
        if (step != Playing || !EditorApplication.isPlaying)
            return;
        try
        {
            if (Script.Count == 0)
                Script.Push(Play());
            while (Script.Count > 0)
            {
                IEnumerator top = Script.Peek();
                if (!top.MoveNext())
                {
                    Script.Pop();
                    continue;
                }
                if (top.Current is IEnumerator nested)
                {
                    Script.Push(nested);
                    continue;
                }
                return;
            }
            EndPlay();
        }
        catch (Exception e)
        {
            Check(false, $"exception (the sweep stops here): {e}");
            try { ScreenCapture.CaptureScreenshot(Path.Combine(Dir, "shots", "exception.png")); } catch { }
            Script.Clear();
            EndPlay();
        }
    }

    private static void EndPlay()
    {
        SessionState.SetInt(StepKey, Finishing);
        EditorApplication.ExitPlaymode();
    }

    private static IEnumerator Play()
    {
        yield return Until(() => Object.FindFirstObjectByType<TitleSceneController>() != null && RunManager.HasInstance, 60f, "the title scene");
        RunManager.Instance.Config.fixedRunSeed = Seed;
        yield return Frames(5);
        yield return Title();
        yield return OfficeDay1();
        yield return HomeNight();
        yield return Until(() => Office.Loaded() && RunManager.Instance.World.day == 2, 90f, "day 2's office");
        yield return Frames(5);
        Note("RunManager.JumpToDay(5): the scanner, the Investigation app and every PC app are in from day 5");
        RunManager.Instance.JumpToDay(5);
        yield return OfficeDay5();
        yield return TitleContinue();
    }

    // ---- Title ----

    private static IEnumerator Title()
    {
        var title = Object.FindFirstObjectByType<TitleUIController>();
        var titlePanel = (GameObject)Field(title, "titlePanel");
        var newRun = (Button)Field(title, "newRunButton");
        var cont = (Button)Field(title, "continueButton");
        yield return Frames(10);
        yield return Shot("title", "title");
        Check(titlePanel.activeInHierarchy && newRun.gameObject.activeInHierarchy && newRun.interactable, "title: New Run is shown and takes clicks");
        Check(!cont.gameObject.activeInHierarchy, "title: Continue is hidden with no save");
        Note("title: the Title has no Settings button; Settings is the PC's Settings app (checked on day 5)");
        newRun.onClick.Invoke();
        yield return Frames(5);
        var adopt = (GameObject)Field(title, "adoptPanel");
        Check(adopt != null && adopt.activeInHierarchy, "title: New Run opens the pet adoption");
        if (adopt == null || !adopt.activeInHierarchy)
            yield break;
        yield return Shot("title", "adopt");
        var input = (TMP_InputField)Field(title, "adoptNameInput");
        input.text = "R2-D2";
        ((Button)Field(title, "adoptButton")).onClick.Invoke();
        yield return Frames(3);
        string why = ((TMP_Text)Field(title, "adoptProblemText")).text;
        Check(adopt.activeInHierarchy && why.Length > 0, $"adopt: a name with digits is refused and says why ('{why}')");
        yield return Shot("title", "adopt_refused");
        ((Button)Field(title, "adoptCatButton")).onClick.Invoke();
        yield return Frames(2);
        Check(!((Button)Field(title, "adoptCatButton")).interactable, "adopt: the cat button shows it is chosen");
        Check(CoatIds(title).SequenceEqual(new[] { "white", "ginger", "slate", "black", "cream" }), $"adopt: the cat offers its five coats ({string.Join(", ", CoatIds(title))})");
        yield return Shot("title", "adopt_cat");
        ((Button)Field(title, "adoptDogButton")).onClick.Invoke();
        yield return Frames(2);
        Check(CoatIds(title).SequenceEqual(new[] { "cream", "tan", "chocolate", "charcoal", "grey" }), $"adopt: the dog offers its five coats ({string.Join(", ", CoatIds(title))})");
        Button charcoal = CoatButton(title, "charcoal");
        Check(charcoal != null && charcoal.interactable, "adopt: the charcoal coat takes a click");
        charcoal?.onClick.Invoke();
        yield return Seconds(0.5f);
        Check(charcoal != null && !charcoal.interactable && CoatButton(title, "cream").interactable, "adopt: the charcoal swatch shows it is chosen, the default one is live again");
        string preview = PetPicture((PetStandIn)Field(title, "adoptPreview"));
        Check(preview == "pet_dog_charcoal_idle", $"adopt: the preview shows the dog in the charcoal coat, idle ('{preview}')");
        yield return Shot("title", "adopt_coat");
        input.text = "Biscuit";
        ((Button)Field(title, "adoptButton")).onClick.Invoke();
        yield return Frames(3);
        WorldState w = RunManager.Instance.World;
        Check(w.pet != null && w.pet.name == "Biscuit" && w.pet.kind == PetKind.Dog && w.pet.coat == "charcoal",

    }

    private static IEnumerator TitleContinue()
    {
        RunManager.Instance.LoadTitleScene();
        yield return Until(() => Object.FindFirstObjectByType<TitleUIController>() != null && SceneManager.GetActiveScene().name == RunManager.Instance.Config.titleSceneName, 60f, "the Title again");
        yield return Frames(10);
        var title = Object.FindFirstObjectByType<TitleUIController>();
        var cont = (Button)Field(title, "continueButton");
        yield return Shot("title", "title_with_save");
        Check(cont.gameObject.activeInHierarchy && cont.interactable, "title: Continue is shown with a save");
        if (!cont.gameObject.activeInHierarchy)
            yield break;
        int day = RunManager.Instance.World.day;
        cont.onClick.Invoke();
        yield return Until(() => SceneManager.GetActiveScene().name != RunManager.Instance.Config.titleSceneName, 60f, "Continue's scene");
        yield return Frames(20);
        Check(RunManager.Instance.World.day >= 1, $"title: Continue resumes the run in {SceneManager.GetActiveScene().name} (day {RunManager.Instance.World.day}, was {day})");
        yield return Shot("title", "continued");
    }

    private static IEnumerator OfficeLoads(int day)
    {
        yield return Until(() => Office.Loaded() && RunManager.Instance.World.day == day && Office.Briefing() != null && Office.Briefing().activeInHierarchy, 90f, $"day {day}'s office and briefing");
        yield return Seconds(1.5f);
    }

    private static IEnumerator OfficeDay1()
    {
        yield return OfficeLoads(1);
        var o = new Office();
        yield return Shot("office", "d1_briefing");
        CheckInside(Office.Briefing().transform as RectTransform, "day 1: the briefing");
        BureauMemo(o, "day 1");
        o.StartShift.onClick.Invoke();
        yield return Seconds(1f);
        yield return Shot("office", "d1_available_start");
        Check(!o.Game.Desk.IsAvailable && o.Ready.Interactable, "day 1: the desk starts paused and the AVAILABLE sign takes clicks");
        yield return KitChecks("office", "day 1 start");
        o.ClickSign();
        Check(o.Game.Desk.IsAvailable, "day 1: a click on AVAILABLE opens the desk");
        yield return Until(() => o.AtDesk, 40f, "day 1's first traveller");
        yield return Seconds(1.5f);
        if (o.Guide.ShowingNow == "step")
        {
            yield return Shot("office", "d1_tutorial");
            ((Button)Field(o.Guide.Prompt, "skipButton")).onClick.Invoke();
            yield return Frames(3);
        }
        yield return Shot("office", "d1_traveller_at_desk");
        yield return KitChecks("office", "day 1 traveller at the desk");
        HelixMonitor(o, "d1");
        CameraSway("d1");

        yield return Wheel(o, "d1");
        yield return InspectRulebook(o);
        yield return PapersOnTheDesk(o, "d1");
        yield return RulebookTabs(o, "d1");
        yield return CityAndBack(o, "d1");
        yield return PullTabs(o, "d1");

        // The first traveller is turned away without evidence (a citation arrives), a dater pressed over a field of the passport; the
        // second is decided with a dater on the booklet's visa page; the third is detained; the rest of the day decided right.
        int decided = 0;
        bool first = true;
        while (true)
        {
            if (!first)
            {
                yield return Until(() => o.AtDesk || o.Results.activeInHierarchy, 40f, "day 1: the next traveller or the shift report");
                if (o.Results.activeInHierarchy)
                    break;
                yield return Seconds(0.6f);
            }
            var c = (CaseInstance)Field(o.Ui, "_currentCase");
            bool accept = !first && c.ShouldAccept;
            if (decided == 2)
                yield return Detain(o, "d1_detain");
            else
                yield return Decide(o, accept, decided < 2 ? $"d1_slot{decided + 1}" : null, decided == 0 ? PressAt.Field : decided == 1 ? PressAt.Hanging : PressAt.Anywhere);
            decided++;
            yield return Seconds(0.5f);
            if (o.CitationPanel.activeInHierarchy)
                yield return CitationSlip(o, $"d1_citation_{decided}");
            else if (first)
                Note("day 1: the first traveller's denial brought no citation slip (a free warning, or the traveller was faulty)");
            first = false;
            if (!string.IsNullOrEmpty(RunManager.Instance.World.endingId))
            {
                Check(false, $"day 1 ended the run ({RunManager.Instance.World.endingId})");
                yield break;
            }
        }
        yield return Seconds(1f);
        yield return Shot("day", "d1_shift_report");
        CheckInside(o.Results.transform as RectTransform, "day 1: the shift report");
        Check(o.Game.Ledger.verdicts.Count == decided && decided == RunManager.Instance.GetCurrentDayPlan().VisitorsCount,
              $"day 1: a full day, start to shift report: every traveller decided ({decided} of {RunManager.Instance.GetCurrentDayPlan().VisitorsCount})");
        o.GoHome.onClick.Invoke();
    }

    /// <summary>The wheel: opened as the player does (the traveller clicked), every choice of its first ring and of the menus they open chosen once, with the speech bubble's place checked (on the screen, off the wheel).</summary>
    private static IEnumerator Wheel(Office o, string tag)
    {
        var inspect = Office.One<DeskInspect>();
        if (inspect.IsOn)
            inspect.SetOn(false);
        inspect.TravellerClicked();
        yield return Seconds(0.6f);
        if (!o.Wheel.IsOpen)
        {
            o.Wheel.Open();
            yield return Seconds(0.6f);
            Check(o.Wheel.IsOpen, $"{tag}: the wheel opens (a click on the traveller did not open it; Open did: {o.Wheel.IsOpen})");
        }
        else
        {
            Check(true, $"{tag}: a click on the traveller opens the wheel");
        }
        yield return Shot("office", $"{tag}_wheel_open");
        yield return KitChecks("office", $"{tag} wheel open");
        object interviewer = Field(o.Ui, "_interview");
        var runner = (DialogRunner)Field(interviewer, "_runner");
        var done = new HashSet<string>();
        int shots = 0;
        for (int guard = 0; guard < 40; guard++)
        {
            DialogChoice next = runner.Choices.FirstOrDefault(x => !done.Contains(x.Id) && x.Kind != DialogChoiceKind.Back && x.Id != InterviewScript.PadChoiceId);
            if (next == null)
            {
                DialogChoice back = runner.Choices.FirstOrDefault(x => x.Kind == DialogChoiceKind.Back);
                if (back == null)
                    break;
                Call(interviewer, "Choose", back.Id);
                yield return Seconds(0.3f);
                continue;
            }
            done.Add(next.Id);
            Call(interviewer, "Choose", next.Id);
            yield return Seconds(0.8f);
            var bubble = (OverlayCallout)Field(o.Wheel, "bubble");
            RectTransform panel = bubble != null ? bubble.Panel : null;
            bool shown = panel != null && panel.gameObject.activeInHierarchy;
            if (shown)
            {
                CheckInside(panel, $"{tag} wheel '{next.Id}': the speech bubble");
                // The wheel is its entries (the ring's own rect is the square round the circle): the bubble covers none of them.
                var entries = ((List<GameObject>)Field(Field(o.Ui, "interactionPanel"), "_spawned")).Where(g => g != null && g.activeInHierarchy).ToList();
                Rect a = ScreenRect(panel);
                List<string> covered = entries.Where(g => ScreenRect((RectTransform)g.transform).Overlaps(a)).Select(g => g.name).ToList();
                if (entries.Count == 0)
                    Note($"{tag} wheel '{next.Id}': the wheel shows no entries after it (the bubble alone)");
                else
                    Check(covered.Count == 0, $"{tag} wheel '{next.Id}': the speech bubble covers none of the wheel's {entries.Count} entries (bubble {Fmt(a)}{List(covered)})");
            }
            Report($"   wheel '{next.Id}' ({next.Kind}) chosen: bubble {(shown ? "shown" : "not shown")} '{o.Wheel.CurrentLine}'");
            if (shots < 5 && shown)
            {
                shots++;
                yield return Shot("office", $"{tag}_wheel_{Safe(next.Id)}");
            }
        }
        Check(done.Count > 0, $"{tag}: the wheel offered choices and each was chosen once ({done.Count}: {string.Join(", ", done)})");
        o.Wheel.Close();
        yield return Seconds(0.3f);
        Check(!o.Wheel.IsOpen, $"{tag}: the wheel closes");
    }

    /// <summary>Inspect mode with the passport's Destination held against today's first rule (day 1: one paper, so a paper and a rulebook row are linked).</summary>
    private static IEnumerator InspectRulebook(Office o)
    {
        var inspect = Office.One<DeskInspect>();
        DeskDocument passport = o.Papers().FirstOrDefault();
        if (passport == null)
        {
            Check(false, "d1 inspect: no paper on the desk");
            yield break;
        }
        if (passport.Zone != DeskZone.Desk)
            Call(o.Desk, "HandlePaperClicked", passport, -1);
        yield return Seconds(0.8f);
        inspect.SetOn(true);
        yield return Seconds(0.4f);
        Check(inspect.IsOn, "d1: inspect mode turns on");
        yield return Shot("office", "d1_inspect_on");
        var c = (CaseInstance)Field(o.Ui, "_currentCase");
        DocumentInstance doc = c.documents[passport.Index];
        int row = doc.fields.FindIndex(f => f.category == ClueCategory.Destination);
        var board = (MatchBoard)Field(inspect, "board");
        int before = board.Log.Count;
        var rows = (Clickable[])Field(o.Rulebook, "rows");
        if (row >= 0 && rows.Length > 0 && rows[0] != null && rows[0].gameObject.activeInHierarchy)
        {
            var compare = (CompareController)Field(inspect, "compare");
            compare?.Clear(); // the wheel's Look > Face leaves the face held
            Call(o.Desk, "HandlePaperClicked", passport, passport.SlotOfRow(row));
            yield return Frames(3);
            bool held = compare != null && compare.Holding;
            rows[0].onClick.Invoke();
            yield return Seconds(0.6f);
            Check(board.Log.Count > before, $"d1 inspect: the passport's Destination (slot {passport.SlotOfRow(row)}, held {held}, zone {passport.Zone}) linked to the rulebook's first rule logs a finding ({before} -> {board.Log.Count}; reading view {((DeskView)Field(o.Desk, "deskView")).IsOn})");
            yield return Shot("office", "d1_inspect_linked");
        }
        else
        {
            Check(false, $"d1 inspect: no Destination field ({row}) or no rule row to link");
        }
        inspect.SetOn(false);
        yield return Frames(3);
    }

    /// <summary>Each paper picked up, dragged, dropped on the desk and read in the reading view; with two papers, they are laid over one another both ways and the one on top must draw on top.</summary>
    private static IEnumerator PapersOnTheDesk(Office o, string tag)
    {
        var view = (DeskView)Field(o.Desk, "deskView");
        Vector3 deskAt = ((DeskSurface)Field(o.Desk, "surface")).transform.position;
        List<DeskDocument> papers = o.Papers();
        foreach (DeskDocument p in papers)
        {
            yield return Until(() => !p.IsSliding, 6f, "a paper at rest");
            Vector3 to = deskAt + new Vector3(0.05f * p.Index, 0f, 0.02f);
            p.Drag.RememberPosition();
            Call(o.Desk, "HandleDragBegan", p.Drag);
            Call(o.Desk, "HandleDragged", p.Drag, to);
            yield return Frames(3);
            p.transform.position = to;
            Call(o.Desk, "HandleDragEnded", p.Drag, to);
            yield return Seconds(0.6f);
            Check(p.Zone == DeskZone.Desk, $"{tag} paper {p.Index}: picked up and dropped on the desk it lies on the desk ({p.Zone})");
            Call(o.Desk, "HandlePaperClicked", p, -1);
            yield return Seconds(0.3f);
            if (view != null)
            {
                view.TiltIn();
                yield return Seconds(0.8f);
                Check(view.IsOn, $"{tag} paper {p.Index}: the reading view tilts in");
                yield return Shot("office", $"{tag}_paper{p.Index}_reading");
                view.Return();
                yield return Seconds(0.8f);
            }
        }
        yield return DrawOrder(o, $"{tag} papers laid out");
        DeskDepth(o, $"{tag} papers laid out");
        if (papers.Count < 2)
        {
            Note($"{tag}: one paper only; the overlap is checked on day 5");
            yield break;
        }
        DeskDocument a = papers[0], b = papers[1];
        Vector3 spot = deskAt + new Vector3(0f, 0f, 0.01f);
        foreach ((DeskDocument top, DeskDocument under) in new[] { (a, b), (b, a) })
        {
            foreach (DeskDocument p in new[] { under, top })
            {
                Vector3 at = spot + (p == top ? new Vector3(0.03f, 0f, 0.02f) : Vector3.zero);
                p.Drag.RememberPosition();
                Call(o.Desk, "HandleDragBegan", p.Drag);
                Call(o.Desk, "HandleDragged", p.Drag, at);
                yield return Frames(2);
                p.transform.position = at;
                Call(o.Desk, "HandleDragEnded", p.Drag, at);
                yield return Seconds(0.5f);
            }
            var stack = (PaperStack)Field(o.Desk, "_stack");
            Check(stack.IndexOf(top.Index) > stack.IndexOf(under.Index), $"{tag}: paper {top.Index} dropped over paper {under.Index} lies above it in the stack ({stack.IndexOf(top.Index)} > {stack.IndexOf(under.Index)})");
            yield return Seconds(0.6f);
            yield return DrawOrder(o, $"{tag} paper {top.Index} over paper {under.Index}");
            yield return Shot("office", $"{tag}_overlap_{top.Index}_over_{under.Index}");
        }
    }

    /// <summary>Every paper on the desk and the folder lie over the desk top, their edges too (Saleh's 1007d playtest: papers clipped into the table): each one's lowest mesh point at least half PaperLayers.DeskClearance over it.</summary>
    private static void DeskDepth(Office o, string when)
    {
        float top = ((DeskSurface)Field(o.Desk, "surface")).transform.position.y;
        var low = new List<string>();
        foreach (DeskDocument p in o.Papers().Where(p => p.Zone == DeskZone.Desk && p.gameObject.activeInHierarchy))
        {
            float lowest = p.Sheet.GetComponentsInChildren<MeshRenderer>(false).Where(r => r.GetComponent<TMP_Text>() == null).Min(r => r.bounds.min.y);
            if (lowest < top + PaperLayers.DeskClearance * 0.5f)
                low.Add($"paper {p.Index} {(lowest - top) * 1000f:0.00} mm");
        }
        DeskRulebook book = o.Rulebook;
        if (book != null && book.gameObject.activeInHierarchy)
        {
            float lowest = book.GetComponentsInChildren<MeshRenderer>(false).Where(r => r.GetComponent<TMP_Text>() == null).Min(r => r.bounds.min.y);
            if (lowest < top + PaperLayers.DeskClearance * 0.5f)
                low.Add($"the folder {(lowest - top) * 1000f:0.00} mm");
        }
        Check(low.Count == 0, $"{when}: every paper and the folder lie over the desk top, edges included{List(low)}");
    }

    /// <summary>On the desk no paper is drawn over one lying above it: of two papers whose sheets overlap, the higher in the stack sits higher (nearer the camera).</summary>
    private static IEnumerator DrawOrder(Office o, string when)
    {
        yield return Seconds(0.5f);
        var stack = (PaperStack)Field(o.Desk, "_stack");
        List<DeskDocument> desk = o.Papers().Where(p => p.Zone == DeskZone.Desk && p.gameObject.activeInHierarchy).ToList();
        int pairs = 0, wrong = 0;
        var bad = new List<string>();
        for (int i = 0; i < desk.Count; i++)
            for (int j = i + 1; j < desk.Count; j++)
            {
                Bounds bi = SheetBounds(desk[i]), bj = SheetBounds(desk[j]);
                if (!(bi.min.x < bj.max.x && bj.min.x < bi.max.x && bi.min.z < bj.max.z && bj.min.z < bi.max.z))
                    continue;
                pairs++;
                DeskDocument hi = stack.IndexOf(desk[i].Index) > stack.IndexOf(desk[j].Index) ? desk[i] : desk[j];
                DeskDocument lo = hi == desk[i] ? desk[j] : desk[i];
                if (hi.Sheet.position.y <= lo.Sheet.position.y)
                {
                    wrong++;
                    bad.Add($"{hi.Index} (stack {stack.IndexOf(hi.Index)}, y {hi.Sheet.position.y:0.0000}) under {lo.Index} (y {lo.Sheet.position.y:0.0000})");
                }
            }
        Check(wrong == 0, $"{when}: no paper is drawn over one lying above it ({pairs} overlapping pairs{(bad.Count > 0 ? ": " + string.Join("; ", bad) : "")})");
    }

    private static Bounds SheetBounds(DeskDocument p)
    {
        Renderer[] rs = p.Sheet.GetComponentsInChildren<Renderer>(false);
        if (rs.Length == 0)
            return new Bounds(p.Sheet.position, Vector3.zero);
        Bounds b = rs[0].bounds;
        foreach (Renderer r in rs)
            b.Encapsulate(r.bounds);
        return b;
    }

    // ---- The stamps: the decision ----

    /// <summary>Where a dater is pressed on the passport: dragged anywhere on it, dragged over one of its printed fields' boxes, dragged onto the booklet's visa page (its stamp area), or clicked where it hangs (it must print on the visa page).</summary>
    private enum PressAt { Anywhere, Field, Visa, Hanging }

    /// <summary>
    /// A traveller decided with the desk machine (the desk machine spec): the
    /// dater rack out, the passport on the desk, the APPROVED or DENIED dater
    /// dragged onto it at <paramref name="at"/> (its mark where it was pressed,
    /// its ink over everything printed on the paper), then the papers handed
    /// back on the counter, which decides (Papers, Please's way).
    /// </summary>
    private static IEnumerator Decide(Office o, bool accept, string shotTag, PressAt at = PressAt.Anywhere)
    {
        int before = o.Game.Ledger.verdicts.Count;
        if (!o.Stamps.BarOut)
            o.Controls.Press(ControlKey.Stamps);
        yield return Seconds(0.6f);
        DeskDocument passport = o.Paper((int)Field(o.Stamps, "_passport"));
        if (passport == null)
        {
            Check(false, "the passport is on the desk to stamp");
            yield break;
        }
        yield return Until(() => !passport.IsSliding, 6f, "the passport at rest");
        if (passport.Zone != DeskZone.Desk)
            Call(o.Desk, "HandlePaperClicked", passport, -1);
        yield return Seconds(0.9f);
        Call(o.Desk, "HandlePaperClicked", passport, -1);
        yield return Frames(2);
        yield return RackClear(o, passport, shotTag);
        Vector3 target = passport.Sheet.TransformPoint(new Vector3(0f, passport.Size.y * 0.2f, 0f));
        Bounds field = default;
        FaceRect visa = default;
        if (at == PressAt.Field)
        {
            var c = (CaseInstance)Field(o.Ui, "_currentCase");
            DocumentInstance doc = c.documents[passport.Index];
            int row = doc.fields.FindIndex(f => f.category == ClueCategory.Destination);
            if (row < 0)
                row = 0;
            if (passport.TryBoundsOf(passport.SlotOfRow(row), out field))
                target = field.center;
            else
                Check(false, $"{shotTag}: the passport's field {row} has a box to press over");
        }
        if (at == PressAt.Visa || at == PressAt.Hanging)
        {
            var form = (PlacedForm)Field(passport, "_form");
            float scale = (float)Field(passport, "_scale");
            visa = StampSpots.Area(form);
            float cx = (visa.XMin + visa.XMax) / 2f, cy = (visa.YMin + visa.YMax) / 2f;
            target = passport.Sheet.TransformPoint(new Vector3((cx - form.Width / 2f) * scale, (form.PageHeight / 2f - cy) * scale, 0f));
        }
        if (at == PressAt.Hanging)
        {
            // A click on a hanging dater presses it where it hangs (DeskStampTray.Press, as PointerHold's quick click does).
            var click = (Clickable)Field(o.Stamps, accept ? "approvedStamp" : "deniedStamp");
            object handle = ((Array)Field(o.Stamps, "_handles")).Cast<object>().First(h => (Clickable)Field(h, "Click") == click);
            Call(o.Stamps, "Press", handle, false);
        }
        else
        {
            yield return DragStamp(o, accept, target);
        }
        yield return Settle(() => !o.Stamps.IsMoving, 5f);
        Check(o.Stamps.HasVerdict, $"the {(accept ? "APPROVED" : "DENIED")} dater {(at == PressAt.Hanging ? "pressed where it hangs" : "dragged onto the passport")} stamps it ({o.Stamps.LastPress})");
        Renderer mark = passport.GetComponentsInChildren<Renderer>(true).FirstOrDefault(r => r.name == "Stamp_1" && r.gameObject.activeSelf);
        if (mark != null && at != PressAt.Anywhere)
            InkOnTop(passport, mark, shotTag);
        if (mark != null && at == PressAt.Field)
        {
            bool over = mark.bounds.min.x < field.max.x && field.min.x < mark.bounds.max.x && mark.bounds.min.z < field.max.z && field.min.z < mark.bounds.max.z;
            Check(over, $"{shotTag}: the dater pressed over the passport's field prints its mark over that field's box (mark {mark.bounds.center}, box {field.center})");
        }
        if (mark != null && (at == PressAt.Visa || at == PressAt.Hanging))
        {
            float scale = (float)Field(passport, "_scale");
            Vector2 page = passport.PagePoint(mark.bounds.center) / scale;
            bool inside = page.x >= visa.XMin && page.x <= visa.XMax && page.y >= visa.YMin && page.y <= visa.YMax;
            Check(inside, $"{shotTag}: the dater pressed on the booklet's visa page prints inside its stamp area (mark at ({page.x:0.00},{page.y:0.00}) of the area ({visa.XMin:0.00},{visa.YMin:0.00})-({visa.XMax:0.00},{visa.YMax:0.00}))");
        }
        if (shotTag != null)
        {
            yield return Shot("office", $"{shotTag}_stamped");
            var view = (DeskView)Field(o.Desk, "deskView");
            if (view != null)
            {
                view.TiltIn();
                yield return Seconds(0.8f);
                yield return Shot("office", $"{shotTag}_stamped_reading");
                view.Return();
                yield return Seconds(0.8f);
            }
        }
        yield return Settle(() => !o.Stamps.IsCarrying, 5f);
        Vector3 onCounter = CounterPoint(o);
        passport.transform.position = onCounter;
        Call(o.Desk, "HandleDragEnded", passport.Drag, onCounter);
        yield return Frames(3);
        Check(o.Game.Ledger.verdicts.Count == before + 1, $"the stamped passport handed back on the counter decides the traveller ({o.Game.Ledger.verdicts.Count - before} verdict)");
    }

    /// <summary>
    /// With the rack out in the reading view (the orchestrator's check from
    /// run 7's contact sheets): the passport's visa page lies under the daters
    /// (its stamp area's middle within 1 cm of the dies' middle), and no part
    /// of the rack covers any of its field boxes or its photo on the screen.
    /// </summary>
    private static IEnumerator RackClear(Office o, DeskDocument paper, string tag)
    {
        bool shots = tag != null;
        tag ??= "a traveller";
        yield return Settle(() => !paper.IsSliding && !o.Stamps.IsMoving, 3f);
        yield return Seconds(0.4f);
        var view = (DeskView)Field(o.Desk, "deskView");
        Camera cam = OfficeRaycaster().eventCamera;
        var rack = (Transform)Field(o.Stamps, "rack");
        var parts = rack.GetComponentsInChildren<Renderer>(false).Select(r => (name: r.name, rect: ScreenBounds(cam, r.bounds))).ToList();
        var fields = new List<(string name, Rect rect)>();
        for (int s = 0; s < paper.SlotCount; s++)
            if (paper.TryBoundsOf(s, out Bounds b))
                fields.Add(($"field {s}", ScreenBounds(cam, b)));
        var photo = (GameObject)Field(paper, "photoSlot");
        Renderer[] photoParts = photo != null && photo.activeInHierarchy ? photo.GetComponentsInChildren<Renderer>(false) : new Renderer[0];
        if (photoParts.Length > 0)
            fields.Add(("photo", ScreenBounds(cam, BoundsOf(photoParts))));
        var covered = new List<string>();
        foreach ((string name, Rect rect) f in fields)
            foreach ((string name, Rect rect) p in parts)
            {
                float w = Mathf.Min(f.rect.xMax, p.rect.xMax) - Mathf.Max(f.rect.xMin, p.rect.xMin), h = Mathf.Min(f.rect.yMax, p.rect.yMax) - Mathf.Max(f.rect.yMin, p.rect.yMin);
                if (w > 2f && h > 2f)
                {
                    covered.Add($"{f.name} {Fmt(f.rect)} under {p.name} {Fmt(p.rect)}");
                    break;
                }
            }
        Check(view != null && view.IsOn && fields.Count > 0 && covered.Count == 0,
              $"{tag}: with the rack out in the reading view it covers none of the passport's {fields.Count} field boxes and photo (reading view {view != null && view.IsOn}){List(covered)}");
        if (paper.TryStampAreaOffset(out Vector3 offset) && o.Stamps.TryStampSpot(out Vector3 dies))
        {
            Vector3 area = paper.transform.position + offset;
            float off = new Vector2(area.x - dies.x, area.z - dies.z).magnitude;
            Check(off < 0.01f, $"{tag}: the passport's stamp area lies at the stamp spot above the drawer ({off * 1000f:0} mm from it)");
        }
        else
        {
            Check(false, $"{tag}: the passport has a stamp area and the rack is laid");
        }
        if (shots)
            yield return Shot("office", $"{tag}_rack_out_reading");
    }

    /// <summary>The dater's ink lies over everything printed on the paper (its own layer, the highest of the paper's parts: PaperLayers.Ink) and draws after the paper's printed layers.</summary>
    private static void InkOnTop(DeskDocument paper, Renderer mark, string tag)
    {
        float inkLift = -mark.transform.localPosition.z;
        var under = new List<string>();
        foreach (Renderer r in paper.Sheet.GetComponentsInChildren<Renderer>(false))
        {
            if (r == mark || r.name.StartsWith("Stamp_") || r.transform.parent != mark.transform.parent)
                continue;
            float lift = -r.transform.localPosition.z;
            int queue = r.sharedMaterial != null ? r.sharedMaterial.renderQueue : 0;
            if (lift >= inkLift || queue > mark.sharedMaterial.renderQueue)
                under.Add($"{r.name} (lift {lift * 1000f:0.0} mm, queue {queue})");
        }
        Check(Mathf.Abs(inkLift - PaperLayers.Ink) < 1e-6f && under.Count == 0,
              $"{tag}: the dater's ink lies over every printed part of the paper (ink lift {inkLift * 1000f:0.0} mm, queue {mark.sharedMaterial.renderQueue}){List(under)}");
    }

    private static PhysicsRaycaster OfficeRaycaster() =>
        Object.FindObjectsByType<PhysicsRaycaster>(FindObjectsInactive.Exclude, FindObjectsSortMode.None).First(r => r.GetType() == typeof(PhysicsRaycaster));

    /// <summary>A world box's rectangle on the screen (its eight corners through <paramref name="cam"/>).</summary>
    private static Rect ScreenBounds(Camera cam, Bounds b)
    {
        float x0 = float.MaxValue, y0 = float.MaxValue, x1 = float.MinValue, y1 = float.MinValue;
        for (int i = 0; i < 8; i++)
        {
            Vector3 corner = new Vector3((i & 1) == 0 ? b.min.x : b.max.x, (i & 2) == 0 ? b.min.y : b.max.y, (i & 4) == 0 ? b.min.z : b.max.z);
            Vector3 s = cam.WorldToScreenPoint(corner);
            x0 = Mathf.Min(x0, s.x); y0 = Mathf.Min(y0, s.y); x1 = Mathf.Max(x1, s.x); y1 = Mathf.Max(y1, s.y);
        }
        return Rect.MinMaxRect(x0, y0, x1, y1);
    }

    private static Bounds BoundsOf(IEnumerable<Renderer> renderers)
    {
        Bounds b = default;
        bool any = false;
        foreach (Renderer r in renderers)
        {
            if (!any) { b = r.bounds; any = true; }
            else b.Encapsulate(r.bounds);
        }
        return b;
    }

    /// <summary>The dater rack's label lip lies low in front of the daters: it covers little of either dater's body on the screen.</summary>
    private static void RackLip(Office o, string tag)
    {
        var rack = (Transform)Field(o.Stamps, "rack");
        Camera cam = OfficeRaycaster().eventCamera;
        Renderer rail = rack != null ? rack.Find("Rail")?.GetComponent<Renderer>() : null;
        if (rail == null && rack != null && rack.GetComponentInChildren<BrassDrawer>(true) != null)
        {
            Note($"{tag}: the daters stand in the brass drawer (main f645987); the built lip is its fallback only");
            return;
        }
        if (rail == null || cam == null)
        {
            Check(false, $"{tag}: the dater rack has its label lip");
            return;
        }
        Rect lip = ScreenBounds(cam, rail.bounds);
        foreach (string name in new[] { "Denied", "Approved" })
        {
            Transform body = rack.Find(name + "/Body");
            if (body == null)
            {
                Check(false, $"{tag}: the {name} dater has a body");
                continue;
            }
            Rect b = ScreenBounds(cam, BoundsOf(body.GetComponentsInChildren<Renderer>(false).Where(r => r.GetComponent<TMPro.TextMeshPro>() == null)));
            float w = Mathf.Max(0f, Mathf.Min(lip.xMax, b.xMax) - Mathf.Max(lip.xMin, b.xMin)), h = Mathf.Max(0f, Mathf.Min(lip.yMax, b.yMax) - Mathf.Max(lip.yMin, b.yMin));
            float share = b.width * b.height > 0f ? w * h / (b.width * b.height) : 1f;

    /// <summary>The cameras' idle sway (Saleh 2026-10-07: off by default, it caused motion sickness): no camera noise at first; turned on it breathes, turned off it stops.</summary>
    private static void CameraSway(string tag)
    {

    }

    /// <summary>The stability monitor (Track P: drawn 2.4 times its art size so the Helix River reads): big enough on the screen, wholly on it.</summary>
    private static void HelixMonitor(Office o, string tag)
    {
        Camera cam = OfficeRaycaster().eventCamera;
        Rect best = default;
        foreach (HelixRiverMonitor m in Object.FindObjectsByType<HelixRiverMonitor>(FindObjectsInactive.Exclude, FindObjectsSortMode.None))

    private static IEnumerator DragStamp(Office o, bool approved, Vector3 target)
    {
        var click = (Clickable)Field(o.Stamps, approved ? "approvedStamp" : "deniedStamp");
        PhysicsRaycaster raycaster = Object.FindObjectsByType<PhysicsRaycaster>(FindObjectsInactive.Exclude, FindObjectsSortMode.None).First(r => r.GetType() == typeof(PhysicsRaycaster));
        Camera cam = raycaster.eventCamera;
        Vector3 from = cam.WorldToScreenPoint(click.transform.TransformPoint(((BoxCollider)click.GetComponent<Collider>()).center * 1.8f)), to = cam.WorldToScreenPoint(target);
        var data = new PointerEventData(EventSystem.current)
        {
            button = PointerEventData.InputButton.Left,
            position = from,
            pointerPressRaycast = new RaycastResult { module = raycaster, gameObject = click.gameObject }
        };
        ExecuteEvents.Execute(click.gameObject, data, ExecuteEvents.beginDragHandler);
        for (int i = 1; i <= 6; i++)
        {
            data.position = Vector3.Lerp(from, to, i / 6f);
            ExecuteEvents.Execute(click.gameObject, data, ExecuteEvents.dragHandler);
            yield return Frames(1);
        }
        ExecuteEvents.Execute(click.gameObject, data, ExecuteEvents.endDragHandler);
        yield return Frames(2);
    }

    /// <summary>Inspect mode with two papers' fields of the same detail linked (a paper against a paper).</summary>
    private static IEnumerator InspectTwoPapers(Office o)
    {
        var inspect = Office.One<DeskInspect>();
        var c = (CaseInstance)Field(o.Ui, "_currentCase");
        List<DeskDocument> papers = o.Papers().Where(p => p.Zone == DeskZone.Desk).ToList();
        (DeskDocument a, int ra, DeskDocument b, int rb) pick = default;
        bool found = false;
        for (int i = 0; i < papers.Count && !found; i++)
            for (int j = 0; j < papers.Count && !found; j++)
            {
                if (i == j)
                    continue;
                List<DocumentField> fa = c.documents[papers[i].Index].fields, fb = c.documents[papers[j].Index].fields;
                for (int x = 0; x < fa.Count && !found; x++)
                {
                    int y = fb.FindIndex(f => f.category == fa[x].category);
                    if (y >= 0 && papers[i].SlotOfRow(x) >= 0 && papers[j].SlotOfRow(y) >= 0)
                    {
                        pick = (papers[i], x, papers[j], y);
                        found = true;
                    }
                }
            }
        if (!found)
        {
            Check(false, $"d5 inspect: no two papers on the desk share a detail ({papers.Count} on the desk)");
            yield break;
        }
        inspect.SetOn(true);
        yield return Seconds(0.4f);
        var board = (MatchBoard)Field(inspect, "board");
        int before = board.Log.Count;
        ((CompareController)Field(inspect, "compare"))?.Clear();
        Call(o.Desk, "HandlePaperClicked", pick.a, pick.a.SlotOfRow(pick.ra));
        yield return Frames(3);
        Call(o.Desk, "HandlePaperClicked", pick.b, pick.b.SlotOfRow(pick.rb));
        yield return Seconds(0.6f);
        Check(board.Log.Count > before, $"d5 inspect: paper {pick.a.Index}'s {c.documents[pick.a.Index].fields[pick.ra].category} linked to paper {pick.b.Index}'s logs a finding ({before} -> {board.Log.Count})");
        yield return Shot("office", "d5_inspect_two_papers");
        yield return KitChecks("office", "d5 inspect linked");
        inspect.SetOn(false);
        yield return Frames(3);
    }

    {

            windows[(string)Field(app, "id")] = (DesktopWindow)Field(app, "window");

                continue;

                Check(false, $"pc: the {id} icon has a window");

            CheckInsideLocal(rect, canvas, $"pc: the {id} window (maximised {win.IsMaximised})");

            if (header != null)
            {
                CheckInsideLocal((RectTransform)header.transform, canvas, $"pc: the {id} window's title bar (reachable)");
                yield return ShotPc($"app_{id}");
            }
            if (header != null && win.IsMaximised)
            {
                Note($"pc: the {id} window opens maximised; a maximised window is not dragged (WindowDrag), so its drag is not tried");
            }
            else if (header != null)
            {
                Vector2 was = rect.anchoredPosition;
                Vector2 from = RectTransformUtility.WorldToScreenPoint(frameCam, header.transform.position);
                var data = new PointerEventData(EventSystem.current) { button = PointerEventData.InputButton.Left, position = from, pressPosition = from };

        yield return CaseBoard(o, canvas, icons, windows);

        IconGrid grid = icons.Grid;

        var spots = Enumerable.Range(0, shown.Count).Select(grid.Spot).ToList();
        bool onGrid = shown.All(i => spots.Any(s => Mathf.Abs(s.x - i.Place.X) < 1f && Mathf.Abs(s.y - i.Place.Y) < 1f))
                      && shown.Select(i => (Mathf.RoundToInt(i.Place.X), Mathf.RoundToInt(i.Place.Y))).Distinct().Count() == shown.Count;
        Check(onGrid, $"pc: Arrange icons lays the {shown.Count} icons on the grid's first spots, one each [{string.Join(" ", shown.Select(i => $"{i.AppId}@{i.Place.X:0},{i.Place.Y:0}"))}]");

            Check(under.Count == 0, $"pc: no icon lies under the waiting strip [{string.Join(", ", under)}]");
        }
        else
        {
            Note("pc: the waiting strip (the desktop's banner) is not shown while a traveller is at the desk");
        }
        yield return ShotPc("arranged");
        o.View.FocusOffice();
        yield return Seconds(1.5f);
        Check(o.View.Current == OfficeView.OfficeFocus, "pc: back to the office");
    }

    /// <summary>The overlay's kit controls at rest: none off the screen, no two overlapping (a control and its own parts excepted).</summary>
    private static IEnumerator KitChecks(string area, string when)
    {
        // At rest: no control's face moving and no popup still flying in (a wheel's pills fly out from its centre and wobble into place).
        yield return Settle(() => Object.FindObjectsByType<UiJuice>(FindObjectsInactive.Exclude, FindObjectsSortMode.None).All(j => !j.Moving)
                                  && Object.FindObjectsByType<UiAppear>(FindObjectsInactive.Exclude, FindObjectsSortMode.None).All(a => !a.Moving), 3f);
        var controls = new List<(UiJuice j, Rect r, Canvas c)>();
        foreach (UiJuice j in Object.FindObjectsByType<UiJuice>(FindObjectsInactive.Exclude, FindObjectsSortMode.None))
        {
            Canvas canvas = j.GetComponentInParent<Canvas>();
            if (canvas == null || canvas.rootCanvas.renderMode != RenderMode.ScreenSpaceOverlay || !Visible(j.transform))
                continue;
            controls.Add((j, ScreenRect((RectTransform)j.transform), canvas.rootCanvas));
        }
        var screen = new Rect(0f, 0f, UnityEngine.Screen.width, UnityEngine.Screen.height);
        List<string> off = controls.Where(x => x.r.xMin < screen.xMin - 1f || x.r.yMin < screen.yMin - 1f || x.r.xMax > screen.xMax + 1f || x.r.yMax > screen.yMax + 1f)
            .Select(x => $"{PathOf(x.j.transform)} {Fmt(x.r)}").ToList();
        Check(off.Count == 0, $"{area} {when}: no kit control off the screen ({controls.Count} shown){List(off)}");
        var overlaps = new List<string>();
        for (int i = 0; i < controls.Count; i++)
            for (int k = i + 1; k < controls.Count; k++)
            {
                Transform a = controls[i].j.transform, b = controls[k].j.transform;
                if (a.IsChildOf(b) || b.IsChildOf(a))
                    continue;
                Rect ra = controls[i].r, rb = controls[k].r;
                float w = Mathf.Min(ra.xMax, rb.xMax) - Mathf.Max(ra.xMin, rb.xMin), h = Mathf.Min(ra.yMax, rb.yMax) - Mathf.Max(ra.yMin, rb.yMin);
                if (w > 1f && h > 1f)
                    overlaps.Add($"{PathOf(a)} x {PathOf(b)} ({w:0}x{h:0} px)");
            }
        Check(overlaps.Count == 0, $"{area} {when}: no two kit controls overlap at rest{List(overlaps)}");
    }

    /// <summary>True when the control draws: its object and every CanvasGroup above it are not see-through.</summary>
    private static bool Visible(Transform t)
    {
        foreach (CanvasGroup g in t.GetComponentsInParent<CanvasGroup>())
            if (g.alpha < 0.01f)
                return false;
        Graphic graphic = t.GetComponent<Graphic>();
        return graphic == null || (graphic.enabled && graphic.color.a > 0.01f) || t.GetComponentsInChildren<Graphic>(false).Any(x => x.enabled && x.color.a > 0.01f);
    }

    private static void CheckInside(RectTransform rt, string what)
    {
        if (rt == null)
        {
            Check(false, $"{what}: missing");
            return;
        }
        Rect r = ScreenRect(rt);
        bool inside = r.xMin >= -1f && r.yMin >= -1f && r.xMax <= UnityEngine.Screen.width + 1f && r.yMax <= UnityEngine.Screen.height + 1f;
        Check(inside, $"{what} lies on the screen ({Fmt(r)} of {UnityEngine.Screen.width}x{UnityEngine.Screen.height})");
    }

    private static void CheckInsideLocal(RectTransform rt, RectTransform canvas, string what)
    {
        if (rt == null)
        {
            Check(false, $"{what}: missing");
            return;
        }
        Rect r = LocalRect(rt, canvas), area = canvas.rect;
        bool inside = r.xMin >= area.xMin - 1f && r.yMin >= area.yMin - 1f && r.xMax <= area.xMax + 1f && r.yMax <= area.yMax + 1f;
        Check(inside, $"{what} lies inside the desktop ({Fmt(r)} of {Fmt(area)})");
    }

    private static Rect ScreenRect(RectTransform rt)
    {
        var c = new Vector3[4];
        rt.GetWorldCorners(c);
        return Rect.MinMaxRect(c.Min(v => v.x), c.Min(v => v.y), c.Max(v => v.x), c.Max(v => v.y));
    }

    private static Rect LocalRect(RectTransform rt, RectTransform canvas)
    {
        var c = new Vector3[4];
        rt.GetWorldCorners(c);
        Vector3[] l = c.Select(canvas.InverseTransformPoint).ToArray();
        return Rect.MinMaxRect(l.Min(v => v.x), l.Min(v => v.y), l.Max(v => v.x), l.Max(v => v.y));
    }

    private static string Fmt(Rect r) => $"({r.xMin:0},{r.yMin:0})-({r.xMax:0},{r.yMax:0})";

    private static string List(List<string> items) => items.Count == 0 ? "" : $": {items.Count} [{string.Join("; ", items.Take(12))}{(items.Count > 12 ? "; ..." : "")}]";

    private static string PathOf(Transform t) => t.parent == null || t.parent.GetComponent<Canvas>() != null && t.parent.parent == null ? t.name : PathOf(t.parent) + "/" + t.name;

    private static string Safe(string s) => new string((s ?? "").Select(ch => char.IsLetterOrDigit(ch) ? ch : '_').ToArray());

    private static IEnumerator Shot(string area, string name)
    {
        Directory.CreateDirectory(System.IO.Path.Combine(Dir, "shots", area));

    }

    private static IEnumerator Capture(string area, string file)
    {
        string path = System.IO.Path.Combine(Dir, "shots", area, file);
        if (File.Exists(path))
            File.Delete(path);
        yield return Frames(2);
        ScreenCapture.CaptureScreenshot(path);
        float end = Time.realtimeSinceStartup + 10f;
        while (!File.Exists(path) && Time.realtimeSinceStartup < end)
            yield return null;
        if (!File.Exists(path))
            Report($"   screenshot {area}/{file} NOT WRITTEN");
    }

    /// <summary>A PC screen: the screen capture (what the frame's glass shows) and the desktop camera's own render beside it.</summary>
    private static IEnumerator ShotPc(string name)
    {
        yield return Shot("pc", name);
        int mask = 1 << OfficeLayers.PcDesktopLayer;
        Camera cam = Object.FindObjectsByType<Camera>(FindObjectsInactive.Include, FindObjectsSortMode.None)
            .Where(c => (c.cullingMask & mask) != 0).OrderBy(c => c.targetTexture != null ? 1 : 0).FirstOrDefault();
        if (cam == null)
            yield break;
        int height = 1080, width = 1440;
        RenderTexture rt = RenderTexture.GetTemporary(width, height, 24);
        RenderTexture old = cam.targetTexture;
        Rect rect = cam.rect;
        bool wasEnabled = cam.enabled;
        cam.targetTexture = rt;
        cam.rect = new Rect(0f, 0f, 1f, 1f);
        cam.enabled = true;
        cam.Render();
        cam.targetTexture = old;
        cam.rect = rect;
        cam.enabled = wasEnabled;
        RenderTexture active = RenderTexture.active;
        RenderTexture.active = rt;
        var tex = new Texture2D(width, height, TextureFormat.RGB24, false);
        tex.ReadPixels(new Rect(0, 0, width, height), 0, 0);
        tex.Apply();
        RenderTexture.active = active;
        File.WriteAllBytes(System.IO.Path.Combine(Dir, "shots", "pc", $"{name}_desktop.png"), tex.EncodeToPNG());
        Object.Destroy(tex);
        RenderTexture.ReleaseTemporary(rt);
        File.AppendAllText(ShotsList, $"pc\t{name} (desktop)\tshots/pc/{name}_desktop.png\n");
    }

    private static IEnumerator Resolution(int width, int height)

        while (!(UnityEngine.Screen.width == width && UnityEngine.Screen.height == height) && Time.realtimeSinceStartup < end)
            yield return null;
        yield return Frames(5);
    }

    private static IEnumerator Frames(int n)
    {
        int target = Time.frameCount + n;
        while (Time.frameCount < target)
            yield return null;
    }

    private static IEnumerator Seconds(float seconds)
    {
        float end = Time.realtimeSinceStartup + seconds;
        while (Time.realtimeSinceStartup < end)
            yield return null;
    }

    private static IEnumerator Settle(Func<bool> done, float seconds)
    {
        float end = Time.realtimeSinceStartup + seconds;
        while (!done() && Time.realtimeSinceStartup < end)
            yield return null;
    }

    private static IEnumerator Until(Func<bool> condition, float timeout, string what)
    {
        float end = Time.realtimeSinceStartup + timeout;
        while (!condition())
        {
            if (Time.realtimeSinceStartup > end)
                throw new TimeoutException($"waited {timeout} s for {what}");
            yield return null;
        }
    }

    private static void Finish()
    {
        PlayModeWindow.SetCustomRenderingResolution((uint)SessionState.GetInt(ViewWidthKey, 1920), (uint)SessionState.GetInt(ViewHeightKey, 1080), ViewName);
        int hadSave = SessionState.GetInt(HadSaveKey, 0);
        for (int i = 0; i < SaveSystem.Files.Count; i++)
        {
            bool had = (hadSave & (1 << i)) != 0;
            if (had && File.Exists(SaveBackup + i))
            {
                File.Copy(SaveBackup + i, SaveSystem.Files[i], true);
                File.Delete(SaveBackup + i);
            }
            else if (!had && File.Exists(SaveSystem.Files[i]))
            {
                File.Delete(SaveSystem.Files[i]);
            }
        }
        foreach (string entry in SessionState.GetString(PrefsKey, string.Empty).Split('\u0002').Where(e => e.Length > 0))
        {
            string[] kv = entry.Split('\u0001');
            if (kv.Length != 3)
                continue;
            if (kv[1] == "absent")
                PlayerPrefs.DeleteKey(kv[0]);
            else if (kv[1] == "float")
                PlayerPrefs.SetFloat(kv[0], float.Parse(kv[2], System.Globalization.CultureInfo.InvariantCulture));
            else
                PlayerPrefs.SetString(kv[0], kv[2]);
        }
        PlayerPrefs.Save();
        Resources.Load<RunConfigSO>(RunManager.ConfigResourcePath).fixedRunSeed = SessionState.GetInt(SeedKey, 0);

        string[] logged = File.Exists(WarningsPath) ? File.ReadAllLines(WarningsPath) : new string[0];
        int errors = logged.Count(l => l.StartsWith("[Error]") || l.StartsWith("[Exception]") || l.StartsWith("[Assert]"));
        Check(errors == 0, $"the log holds no error or exception ({errors} of {logged.Length} lines)");

        // The contact sheets (one per area): the python helper beside this file's committed text.
        try
        {
            string script = System.IO.Path.Combine(Directory.GetCurrentDirectory(), "tools", "audit", "qa_contact_sheet.py");
            var p = System.Diagnostics.Process.Start(new System.Diagnostics.ProcessStartInfo("python", $"\"{script}\" \"{Dir}\"")
            {
                UseShellExecute = false,
                CreateNoWindow = true,
                RedirectStandardOutput = true,
                RedirectStandardError = true
            });
            string output = p.StandardOutput.ReadToEnd() + p.StandardError.ReadToEnd();
            p.WaitForExit(120000);
            Report("contact sheets: " + output.Trim().Replace("\n", " | "));
        }
        catch (Exception e)
        {
            Report($"contact sheets: not made ({e.Message}); run python tools/audit/qa_contact_sheet.py \"{Dir}\"");
        }

        Report($"passes={SessionState.GetInt(PassesKey, 0)} fails={SessionState.GetInt(FailsKey, 0)}");
        SessionState.SetInt(StepKey, 0);
        EditorApplication.update -= Tick;
        Application.logMessageReceived -= OnLog;
        string scene = SessionState.GetString(SceneKey, string.Empty);
        if (!string.IsNullOrEmpty(scene) && File.Exists(scene))
            EditorSceneManager.OpenScene(scene, OpenSceneMode.Single);
        else
            EditorSceneManager.NewScene(NewSceneSetup.EmptyScene, NewSceneMode.Single);
        Report($"done {DateTime.Now:s}");
    }

    private static void OnLog(string message, string stack, LogType type)
    {
        if (type == LogType.Log)
            return;
        if (stack != null && stack.Contains("UnityEditor.Search"))
            return;
        if (message.StartsWith("Account API did not become accessible"))
            return;
        try { File.AppendAllText(WarningsPath, $"[{type}] {message}\n"); } catch (IOException) { }
    }

    /// <summary>The adoption panel's coat swatches' ids, in order.</summary>
    private static List<string> CoatIds(TitleUIController title) =>
        ((IEnumerable<(Button button, string coat)>)Field(title, "_coatSwatches")).Select(s => s.coat).ToList();

    /// <summary>The adoption panel's swatch of <paramref name="coat"/> (null when the row lacks it).</summary>
    private static Button CoatButton(TitleUIController title, string coat) =>
        ((IEnumerable<(Button button, string coat)>)Field(title, "_coatSwatches")).FirstOrDefault(s => s.coat == coat).button;

    /// <summary>The art a PetStandIn draws (its picture's sprite name), or null when it draws the code stand-in.</summary>
    private static string PetPicture(PetStandIn view)
    {
        Transform picture = view != null ? view.transform.Find("Figure/Picture") : null;
        Image image = picture != null ? picture.GetComponent<Image>() : null;
        return image != null && image.sprite != null ? image.sprite.name : null;
    }

    private static object Field(object target, string name)
    {
        for (Type t = target.GetType(); t != null; t = t.BaseType)
        {
            FieldInfo f = t.GetField(name, BindingFlags.Public | BindingFlags.NonPublic | BindingFlags.Instance);
            if (f != null)
                return f.GetValue(target);
        }
        throw new MissingFieldException(target.GetType().Name, name);
    }

    private static object Call(object target, string name, params object[] args)
    {
        MethodInfo m = target.GetType().GetMethods(BindingFlags.Public | BindingFlags.NonPublic | BindingFlags.Instance)
            .First(x => x.Name == name && x.GetParameters().Length == args.Length);
        return m.Invoke(target, args);
    }

    private static void Report(string line) => File.AppendAllText(ReportPath, line + "\n");

    private static void Note(string line) => Report("NOTE " + line);

    private static void Check(bool ok, string what)
    {
        Report((ok ? "PASS " : "FAIL ") + what);
        string key = ok ? PassesKey : FailsKey;
        SessionState.SetInt(key, SessionState.GetInt(key, 0) + 1);
    }
