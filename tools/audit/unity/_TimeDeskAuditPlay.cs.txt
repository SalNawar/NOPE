// Audit baseline automation (kept as text in tools/audit/unity; copied into Assets/Editor as .cs only to run it, never committed there): the Phase 0 scripted
// play-through golden master. Title -> New Run (fixed seed 12345) -> days 1-6
// in the art office RunConfig names (the anime hall) with the gameplay layer,
// Home between them, up to day 7's morning paper. Each shift: one click on the
// AVAILABLE sign after Start Shift, then the travellers come one after another
// (day 1 also pauses the desk at its second traveller, checks nobody new is
// called after them, and resumes). Every traveller: the interview (every request, the papers menu and its request groups
// included, so a rich tourist gives the honest missing-form replies and a poor
// tourist hands over the waiver and the proof; the desk's waiver pad is left
// unused (its answer is logged: offering it would spend shift minutes and cure
// waiver faults); every question of the ask menu;
// slot 1 also runs its first dialog to the end), the evidence for each tell of
// a deviation fault through the real compare path (two papers that disagree
// through the cross proof first; a place tell against the claimed place's book
// row, else the row it really belongs to (the present's for a smuggler or a
// fake displaced person); a record tell against the traveller's own record
// row; a worn tell through Look against the Costume Guide; a forged seal
// against its office's seal in the Seal Register and someone else's photo
// against the traveller's face (Look > Their face) on the PC workbench: the
// paper's box clicked in its scanned copy, the register's seal clicked beside
// it, the findings logging "Seal incorrect" / "Data differs" as evidence with
// the labelled line; the first accepted traveller whose papers show a seal and
// a photo has both held against their truths there too: "Matching data", and
// another office's seal "Another office", nothing logged), then a scripted
// verdict: the right one (deny a deviation fault with its evidence, deny a
// directive fault (a closed destination, a wrong departure date, an expired
// paper) after holding the broken rule against the value it is about on the
// PC workbench (MatchBoard, RuleChecks), accept the rest), except four planned mistakes (an honest
// traveller denied on day 1, a deviation fault accepted on day 2, one denied
// without evidence on day 3, a directive fault accepted on day 4; never a
// visual lie, so the first forged seal or swapped photo is proven). During
// each shift, order at the PC (the Orders app's rule, OrderBook.Order)
// Interview Protocols, then the Near East Speech translator (it needs the
// Protocols delivered), at most one a day, once the wallet holds its price and
// 20 cr more; it arrives the next morning. The re-paced days (wave 5 track C,
// Papers Please lessons 4-6 and D7) are checked on the way: each day's
// briefing opens with its bulletin (days 1-5 each have one), every traveller
// carries only the day's issued papers (day 1: the visa alone) and the PC
// follows them (the shelf's papers, the Citizen Account's forms on file, the
// guided steps shown: no Books before day 7), every
// citation slip names its rule and a value, and every shift report shows
// tonight's bills and the wallet after them. Home: treat any family member at
// condition 2+ (the shop step is skipped: Home stocks no upgrade; never the
// slot machine: its draw is unseeded).
// RunShots plays the same run and also takes the screenshots of the redesign's
// states at 1920x1080 and 1280x720 (<tag>/shots/: each day's paper and shift
// report, each traveller kind's papers held and wheel, the app's documents
// (driven as the clerk does on the PC workbench: a shelf chip opens its
// document on the target side, a step puts its pair up), each form's scanned
// copy beside the Seal Register (the document design spec: every document
// type on the PC; the register from the shelf's books once the seal check
// has arrived), each seal or photo proven on the workbench, the two panes
// and one, the search drawer and its pinned and recent items, each day's
// first traveller's Citizen Account beside their papers with the guided
// steps the day shows (ramp_d<day>), each lie kind caught in the findings
// and on the office strip, each citation reason, and after the transcript's END a
// shots-only day 7 with the Auto-Feed and Analysis scanners set on the desk);
// its transcript must equal RunA's (screenshots never change the run).
// Driven by direct calls from EditorApplication.update in the persistent
// editor (job runner); never exits the editor. The transcript holds only
// deterministic content: <out root>/<tag>/play_transcript.txt (+ saves/ and a
// report; see Root).

    static _TimeDeskAuditPlay()
    {
        if (SessionState.GetInt(StepKey, 0) == 0)
            return;
        EditorApplication.update += Tick;
        Application.logMessageReceived += OnLog;
    }

    public static void RunA() => Begin("playA", false);
    public static void RunB() => Begin("playB", false);

    /// <summary>The same play with the screenshots of the redesign's states (tag playShots); its transcript must equal RunA's up to END.</summary>
    public static void RunShots() => Begin("playShots", true);

    private static void Begin(string tag, bool shots)
    {
        SessionState.SetString(TagKey, tag);
        SessionState.SetBool(ShotsKey, shots);
        Directory.CreateDirectory(Path.Combine(Dir, "saves"));
        foreach (string f in Directory.GetFiles(Dir, "*", SearchOption.AllDirectories))
            File.Delete(f);
        if (shots)
            Directory.CreateDirectory(ShotsDir);
        File.WriteAllText(ReportPath, $"start {DateTime.Now:s} tag={tag} shots={shots}\n");
        File.WriteAllText(TranscriptPath, string.Empty);
        File.WriteAllText(WarningsPath, string.Empty);

        // The player's save slot (the save, its temp file and its backup) is set aside and restored at the end.
        int hadSave = 0;
        for (int i = 0; i < SaveSystem.Files.Count; i++)
        {
            if (!File.Exists(SaveSystem.Files[i]))
                continue;
            hadSave |= 1 << i;
            File.Copy(SaveSystem.Files[i], SaveBackup + i, true);
            File.Delete(SaveSystem.Files[i]);
        }
        SessionState.SetInt(HadSaveKey, hadSave);
        SessionState.SetString(PrefKey, PlayerPrefs.HasKey(UiLanguagePref) ? PlayerPrefs.GetString(UiLanguagePref) : "<absent>");
        SessionState.SetString(MotionKey, PlayerPrefs.HasKey(MotionPref) ? PlayerPrefs.GetString(MotionPref) : "<absent>");
        PlayerPrefs.DeleteKey(UiLanguagePref);
        PlayerPrefs.DeleteKey(MotionPref);

        // The desktop's per-player preferences (icons, split, tab order, zoom, sidebar, steps) start at their defaults and are restored at the end.
        var desktop = new List<string>();
        foreach (string key in DesktopPrefKeys())
        {
            desktop.Add(key + "\u0001" + (PlayerPrefs.HasKey(key) ? PlayerPrefs.GetString(key) : "<absent>"));
            PlayerPrefs.DeleteKey(key);
        }
        SessionState.SetString(DesktopPrefsKey, string.Join("\u0002", desktop));

        // The fixed seed is set on the run's config in play mode (in memory only, never saved; restored at the end).
        SessionState.SetInt(SeedKey, Resources.Load<RunConfigSO>(RunManager.ConfigResourcePath).fixedRunSeed);

        PlayModeWindow.GetRenderingResolution(out uint width, out uint height);
        SessionState.SetInt(ViewWidthKey, (int)width);
        SessionState.SetInt(ViewHeightKey, (int)height);
        PlayModeWindow.SetCustomRenderingResolution(1920, 1080, ViewName);

        SessionState.SetInt(FailsKey, 0);
        SessionState.SetInt(StepKey, Playing);
        SessionState.SetString(SceneKey, SceneManager.GetActiveScene().path);
        EditorSceneManager.OpenScene("Assets/Scenes/TitleScene.unity", OpenSceneMode.Single);
        EditorApplication.update += Tick;
        Application.logMessageReceived += OnLog;
        EditorApplication.EnterPlaymode();
    }

    /// <summary>The PlayerPrefs keys of the desktop's preferences (DesktopPreferences' key constants).</summary>
    private static IEnumerable<string> DesktopPrefKeys() =>
        typeof(DesktopPreferences).GetFields(BindingFlags.Static | BindingFlags.NonPublic | BindingFlags.Public)
            .Where(f => f.IsLiteral && f.FieldType == typeof(string))
            .Select(f => (string)f.GetRawConstantValue())
            .Where(v => v.StartsWith("TimeDesk."));

    private static void Tick()
    {
        int step = SessionState.GetInt(StepKey, 0);
        if (step == Finishing)
        {
            if (!EditorApplication.isPlaying)
                Finish();
            return;
        }
        if (step != Playing || !EditorApplication.isPlaying)
            return;

        try
        {
            if (Script.Count == 0)
                Script.Push(Play());
            while (Script.Count > 0)
            {
                IEnumerator top = Script.Peek();
                if (!top.MoveNext())
                {
                    Script.Pop();
                    continue;
                }
                if (top.Current is IEnumerator nested)
                {
                    Script.Push(nested);
                    continue;
                }
                return;
            }
            EndPlay();
        }
        catch (Exception e)
        {
            Check(false, $"exception: {e}");
            if (ShotsOn)
                try { ScreenCapture.CaptureScreenshot(Path.Combine(ShotsDir, $"diag_exception_{SessionState.GetInt(FailsKey, 0)}.png")); } catch { }
            Script.Clear();
            EndPlay();
        }
    }

    private static void EndPlay()
    {
        SessionState.SetInt(StepKey, Finishing);
        EditorApplication.ExitPlaymode();
    }

    private static IEnumerator Play()
    {
        MistakesMade.Clear();
        Seen.Clear();
        Tally.Clear();
        DeniedFaces.Clear();
        Decided.Clear();
        _ended = false;
        yield return Until(() => Object.FindFirstObjectByType<TitleSceneController>() != null && RunManager.HasInstance, 60f, "the title scene");
        RunManager.Instance.Config.fixedRunSeed = Seed;
        yield return Frames(5);
        Line($"RUN fixed seed {Seed}; title scene {SceneManager.GetActiveScene().name}");
        var title = Object.FindFirstObjectByType<TitleUIController>();
        ((Button)Field(title, "newRunButton")).onClick.Invoke();

        for (int day = 1; day <= LastDay && !_ended; day++)
        {
            yield return OfficeDay(day);
            if (_ended)
                break;
            yield return HomeNight(day);
        }

        // The run ends at night 15's sleep: the day-15 ending on the Title, with its END OF DEMO card.
        WorldState end = RunManager.Instance.World;
        Check(_ended && end.day == LastDay, $"the run plays days 1-{LastDay} and ends at night {LastDay}'s sleep (ended={_ended}, day {end.day}, ending '{end.endingId}')");
        if (_ended)
            yield return EndingScreen(end.endingId);
        Line("END");

        // The run's coverage (report only): the four planned mistakes, each kind, lie and fault met.
        Check(MistakesMade.Count == 4, $"the four planned mistakes were made [{string.Join(", ", MistakesMade)}]");
        foreach (KeyValuePair<string, int> t in Tally)
            Report($"COVER {t.Key}: {t.Value}");
    }

    /// <summary>The ending screen on the Title: the day-15 ending's world page (the endings spec E0: its heading, the world's four outcomes listed plainly, and the END OF DEMO card set apart, EndingSO.closingCard), else a failure's ending panel; each as shown.</summary>
    private static IEnumerator EndingScreen(string endingId)
    {
        TitleUIController ui = null;
        bool Shown(string panel) => ((GameObject)Field(ui, panel)).activeInHierarchy;
        yield return Until(() => (ui = Object.FindFirstObjectByType<TitleUIController>()) != null && (Shown("worldPanel") || Shown("endingPanel")), 60f, "the ending screen on the Title");
        yield return Frames(3);
        bool world = Shown("worldPanel");
        string title = ((TMP_Text)Field(ui, world ? "worldTitleText" : "endingTitleText")).text;
        string body = ((TMP_Text)Field(ui, world ? "worldOutcomesText" : "endingBodyText")).text;
        TMP_Text cardText = world ? (TMP_Text)Field(ui, "worldCardText") : null;
        string card = cardText != null && cardText.gameObject.activeInHierarchy ? cardText.text : "";
        Line($"== ENDING SCREEN '{endingId}' ({(world ? "the world page" : "the ending panel")}): {title}");
        foreach (string l in Lines(body))
            Line("   | " + l);
        foreach (string l in Lines(card))
            Line("   [card] " + l);
        Check(world, $"the day-15 ending shows the world page with the world's outcomes ('{endingId}')");
        if (world)
        {
            // The endings spec E1-E2: every factor's question with its answer as the run left it (ContentLibrarySO.WorldOutcomes), never a number.
            foreach (OutcomeLine o in RunManager.Instance.Library.WorldOutcomes(RunManager.Instance.World, RunManager.Instance.Config.gameConfig))
                Check(body.Contains(o.Question) && !string.IsNullOrWhiteSpace(o.Answer) && body.Contains(o.Answer), $"the world page answers '{o.Question}' with '{o.Answer}'");
            Check(!body.Contains("%"), "the world page shows no percentage (no pull, weight or score)");
        }
        Check(card.Contains("END OF DEMO"), $"the demo ends on the END OF DEMO card under the outcomes ('{endingId}', shown {card.Contains("END OF DEMO")})");
        if (ShotsOn)
            yield return Shots($"ending_{endingId}");
    }

    private static IEnumerator OfficeLoads(int day)
    {
        yield return Until(() => Office.Loaded() && RunManager.Instance.World.day == day && Office.Briefing() != null && Office.Briefing().activeInHierarchy, 90f, $"day {day}'s office and briefing");
        yield return Frames(3);
        Scene art = OfficeScenes.ArtScene, gameplay = SceneManager.GetSceneByName(OfficeScenes.GameplaySceneName);
        Check(art.IsValid() && gameplay.IsValid() && gameplay.isLoaded && SceneManager.GetActiveScene() == art && art.name == RunManager.Instance.Config.officeSceneName,
              $"day {day}: the art office RunConfig names ({art.path}) is active with the gameplay layer ({gameplay.path})");
    }

    private static IEnumerator OfficeDay(int day)
    {
        yield return OfficeLoads(day);
        var o = new Office();
        Line($"== DAY {day}");
        RecordBriefing();
        RecordWorld();
        RecordPortals(o);
        DayPlanSO paced = RunManager.Instance.GetCurrentDayPlan();
        string brief = ((TMP_Text)Field(Office.One<DayFlowUIController>(), "briefingBodyText")).text;
        Check(day > 5 || !string.IsNullOrWhiteSpace(paced.Bulletin), $"day {day}: days 1-5 each announce their one new paper or check in a bulletin ('{paced.Bulletin}')");
        Check(string.IsNullOrWhiteSpace(paced.Bulletin) || (brief.StartsWith(UiText.Get("briefing.bulletinHeader")) && brief.Contains(paced.Bulletin.Trim())),
              $"day {day}: the briefing opens with the day's bulletin");
        WorldHeadlines(day);
        PaperNames(day);
        if (ShotsOn)
            yield return Shots($"d{day}_briefing");
        o.StartShift.onClick.Invoke();
        yield return Frames(2);

        // The AVAILABLE sign (Saleh 2026-09-30): the desk starts paused; one click and the travellers come one after another.
        Check(!o.Game.Desk.IsAvailable && o.Ready.Interactable, $"day {day}: the desk starts paused and the AVAILABLE sign takes clicks");
        OrderAtThePc(day);
        o.ClickSign();
        Check(o.Game.Desk.IsAvailable, $"day {day}: one click turns the desk available");

        int presented = 0;
        bool paused = false;
        while (true)
        {
            yield return Until(() => o.AtDesk || o.Results.activeInHierarchy || _ended, 30f, $"day {day}: the next traveller or the shift report");
            if (o.Results.activeInHierarchy || _ended)
                break;
            presented++;
            yield return Seconds(0.4f);

            // Day 1: pause while the second traveller is at the desk; they are finished normally.
            if (day == 1 && presented == 2)
            {
                o.ClickSign();
                paused = true;
                Check(!o.Game.Desk.IsAvailable && o.AtDesk, "day 1: a second click pauses the desk, the traveller at it stays");
            }

            yield return Traveller(o, day);
            if (_ended)
                break;

            if (paused)
            {
                // Nobody new while paused (the clock runs on), until the sign is clicked again.
                yield return Until(() => (o.Game.Desk.IsWaiting && !o.Game.Desk.IsDeparting) || o.Results.activeInHierarchy, 30f, "day 1: the next traveller waiting while paused");
                float clockBefore = o.Clock.Clock.CurrentMinute;
                yield return Seconds(2f);
                Check(!o.AtDesk && o.Game.Desk.IsWaiting, "day 1: paused, nobody new is called after the traveller at the desk");
                Check(o.Clock.Clock.CurrentMinute > clockBefore, $"day 1: the shift clock keeps running on a break ({clockBefore:0.0} -> {o.Clock.Clock.CurrentMinute:0.0})");
                o.ClickSign();
                Check(o.Game.Desk.IsAvailable && o.AtDesk, "day 1: a third click resumes and calls the waiting traveller at once");
                paused = false;
            }
            OrderAtThePc(day);
        }

        if (_ended)
            yield break;
        yield return Frames(3);
        Line($"-- day {day} shift report ({presented} presented)");
        Line("   " + ((TMP_Text)Field(o.DayFlow(), "resultsTitleText")).text);
        string report = ((TMP_Text)Field(o.DayFlow(), "resultsBodyText")).text;
        foreach (string l in Lines(report))
            Line("   | " + l);
        Check(report.Contains(UiText.Get("results.row.bills")) && report.Contains(UiText.Get("results.row.after")) && report.Contains(UiText.Get("results.row.net")),
              $"day {day}: the shift report shows the net, tonight's bills and the wallet after them");
        ShiftLedger ledger = o.Game.Ledger;
        Line($"   ledger: {ledger.verdicts.Count} verdicts, correct {ledger.CorrectCount}, wrong {ledger.WrongCount}, pay {ledger.TotalPay}, penalties {ledger.TotalPenalties}, stability {ledger.TotalStabilityDelta.ToString("0.###", System.Globalization.CultureInfo.InvariantCulture)}, unproven {ledger.UnprovenDenialCount}");
        Line($"   debt relief: instalment {ledger.debtInstalment}, owed {ledger.debtOwed}, paid so far {RunManager.Instance.World.clerkDebtPaid}");
        Line($"   strandings: {ledger.strandedCount} stranded, fines {ledger.strandingFines}, stability {ledger.strandingStabilityDelta.ToString("0.##", System.Globalization.CultureInfo.InvariantCulture)}");
        foreach (StrandingRecord s in RunManager.Instance.World.history.strandingLog.Where(s => s != null && s.day == day))
        {
            Line($"   stranded: '{s.travellerName}' in {s.placeLabel}: {s.fate}, waivered {s.waivered}, fine {s.fine}, paper '{s.line}'");
            Count($"fate {s.fate}");
        }
        foreach (DialogOutcome d in ledger.dialogOutcomes)
            Line($"   dialog outcome: {JsonUtility.ToJson(d)}");
        Check(ledger.verdicts.Count == RunManager.Instance.GetCurrentDayPlan().VisitorsCount,
              $"day {day}: every traveller in the queue was decided before closing ({ledger.verdicts.Count}/{RunManager.Instance.GetCurrentDayPlan().VisitorsCount})");
        if (ledger.strandedCount > 0)
            Count("stranded");
        RecordSave($"day{day}_shift");
        if (ShotsOn)
            yield return Shots($"d{day}_shift_report");
        o.GoHome.onClick.Invoke();
    }

    /// <summary>The day's order, if any: the first of the plan's upgrades that can be ordered now with 20 cr to spare (one a day), through the Orders app's rule.</summary>
    private static void OrderAtThePc(int day)
    {
        WorldState w = RunManager.Instance.World;
        ContentLibrarySO lib = RunManager.Instance.Library;
        if (w.orders.Any(e => e.orderedDay == day))
            return;
        foreach (string id in new[] { "interview_protocols", "tr_near_east_spoken" })
        {
            UpgradeSO upgrade = lib.GetUpgradeById(id);
            if (upgrade == null || OrderBook.StateOf(w, lib, upgrade) != OrderState.Orderable || w.money < OrderBook.Price(w, lib, upgrade) + 20)
                continue;
            bool placed = OrderBook.Order(w, lib, upgrade);
            Line($"   ordered {id} at the PC: placed={placed} money={w.money} arrives day {day + 1}");
            return;
        }
    }

    private static IEnumerator Traveller(Office o, int day)
    {
        var c = (CaseInstance)Field(o.Ui, "_currentCase");
        int slot = (int)Field(o.Game, "_activeCaseIndex1Based");
        List<DocumentField> fields = c.documents.SelectMany(d => d.fields).ToList();
        string tells = string.Join(",", fields.Where(f => f.isAnachronism).Select(f => $"{f.category}/Papers").Distinct()
            .Concat(c.answers.Where(a => a.isTell).Select(a => $"{a.category}/Answer"))
            .Concat(c.look != null ? c.look.Garments.Where(g => g.IsTell).Select(g => $"{g.Slot}/Dress") : Enumerable.Empty<string>()));
        string recordTells = string.Join(",", c.recordTells.Select(t => $"{t.Category}@{t.Document}"));
        CitizenAccount a = c.account;
        Line($"-- d{day} slot {slot}: '{c.visitorDisplayName}' kind={c.kind} claims '{c.originLabel}' role={(c.archetype != null ? c.archetype.displayName : "-")} gender={c.gender} liar={c.IsLiar} lie={(c.lie.HasValue ? c.lie.Value.ToString() : "-")} home='{c.HomeLabel}' costume={c.costumeFault} directive={c.directiveFault} fault='{c.FaultReason}' shouldAccept={c.ShouldAccept} premade={(c.legendarySource != null ? c.legendarySource.id : "none")} tells=[{tells}] recordTells=[{recordTells}]");
        if (a != null)
            Line($"   account: {a.CitizenId} {a.Status} debt {a.Debt} transponder '{a.Transponder}' {a.TransponderClass} waiver {a.WaiverNo ?? "none"} proof {a.ProofForm ?? "none"} pad {c.waiverPadReply} waivered {c.Waivered}");
        if (c.returning != null)
        {
            // Wave 5, lesson 9: a traveller turned away comes back, same name and face, corrected or with a new story.
            Line($"   returning: turned away on day {c.returning.deniedDay} ({c.returning.story}), window days {c.returning.fromDay}-{c.returning.untilDay}; intro '{c.introLine}'");
            Count($"returning {c.returning.story}");
            Check(DeniedFaces.TryGetValue(c.returning.caseSeed, out string face) && face == Face(c),
                  $"d{day} slot {slot}: '{c.visitorDisplayName}' comes back with the face they were turned away with ({face ?? "never denied"} / {Face(c)})");
            if (c.returning.story == ReturnStory.Corrected)
                Check(!c.HasDeviationFault && !c.HasDirectiveFault, $"d{day} slot {slot}: corrected papers carry no fault ({FaultKey(c)})");
            if (ShotsOn && First("returning"))
            {
                yield return Seconds(1.2f);
                yield return Shots($"d{day}_returning_desk");
                yield return AppShots(o, c, $"d{day}_returning_transcript", AppTab.Transcript, AppTab.Records);
            }
        }
        Line($"   papers: {string.Join(" ; ", c.documents.Select(d => (d.template != null ? d.template.formNumber + " " + d.template.displayName : "?") + ": " + string.Join(", ", d.fields.Select(f => $"{f.label}={f.value}{(f.isAnachronism ? "[TELL]" : "")}"))))}");
        if (c.look != null)
            Line($"   look: {c.look.Describe()} | {string.Join("; ", c.look.Garments.Select(g => $"{g.Slot}:{g.Label}{(g.IsTell ? "[TELL]" : "")}"))}");
        Count($"kind {c.kind}");
        DayPlanSO issuing = RunManager.Instance.GetCurrentDayPlan();
        Check(c.documents.All(d => d.template != null && issuing.Issues(d.template.formNumber)),
              $"d{day} slot {slot}: the traveller carries only the day's issued papers [{string.Join(",", c.documents.Select(d => d.template != null ? d.template.formNumber : "?"))}] of [{string.Join(",", issuing.Papers)}]");
        if (day == 1)
            Check(c.documents.All(d => d.template != null && d.template.formNumber == Directives.Visa), $"d1 slot {slot}: day 1's traveller carries the Leisure Visa alone");
        if (c.lie.HasValue)
            Count($"lie {c.lie.Value}");
        if (c.costumeFault != CostumeError.None)
            Count($"costume {c.costumeFault}");
        if (c.HasDirectiveFault)
            Count($"directive {c.directiveFault}");
        if (c.legendarySource != null)
            Count("premade");

        // The decision (planned mistakes first).
        string plan;
        bool accept;
        bool gatherEvidence;
        if (day == 1 && slot >= 2 && c.ShouldAccept && Once("d1-deny-honest"))
        {
            plan = "MISTAKE deny an honest traveller";
            accept = false;
            gatherEvidence = false;
        }
        else if (day == 2 && c.HasDeviationFault && !Visual(c) && Once("d2-accept-liar"))
        {
            plan = "MISTAKE accept a deviation fault";
            accept = true;
            gatherEvidence = false;
        }
        else if (day == 3 && c.HasDeviationFault && !Visual(c) && Once("d3-unproven-denial"))
        {
            plan = "MISTAKE deny a deviation fault without evidence";
            accept = false;
            gatherEvidence = false;
        }
        else if (day == 4 && c.HasDirectiveFault && Once("d4-accept-violator"))
        {
            plan = $"MISTAKE accept a directive fault ({c.directiveFault})";
            accept = true;
            gatherEvidence = false;
        }
        else if (c.HasDeviationFault)
        {
            plan = "prove the tells, deny";
            accept = false;
            gatherEvidence = true;
        }
        else
        {
            plan = c.ShouldAccept ? "accept" : $"deny (a directive: {c.directiveFault})";
            accept = c.ShouldAccept;
            gatherEvidence = false;
        }

        // The interview: every request, every question (slot 1 also runs its first dialog).
        yield return Interview(o, c, slot == 1);
        CheckRamp(o, c, day, slot);
        if (ShotsOn)
            yield return KindShots(o, c, day);
        if (ShotsOn && First($"ramp d{day}"))
            yield return AppShots(o, c, $"ramp_d{day}", AppTab.Records, AppTab.Documents);

        if (accept && !plan.StartsWith("MISTAKE"))
            yield return VisualProbe(o, c, day, slot);
        if (gatherEvidence)
        {
            yield return Evidence(o, c, day, slot);
            if (ShotsOn)
                yield return CaughtShots(o, c);
            yield return Confront(o, c, day, slot);
        }
        ((CompareController)Field(o.Ui, "compareController")).Clear();
        if (ShotsOn && c.HasDirectiveFault && First($"rules {c.directiveFault}"))
            yield return AppShots(o, c, $"rules_{c.directiveFault}", AppTab.Rules, AppTab.Documents);

        var runner = (DialogRunner)Field(Field(o.Ui, "_interview"), "_runner");
        foreach (DialogLine l in runner.Transcript)
            Line($"   [{l.Speaker}] {l.Text}{(l.IsAnswer ? $" <{l.Category}={l.Value}{(l.IsTell ? " TELL" : "")}>" : "")}");
        var log = (DiscrepancyLog)Field(Field(o.Ui, "_evidence"), "_discrepancies");
        foreach (Discrepancy d in log.Items)
            Line($"   evidence: {d.category} {d.provedBy} from {d.source} value='{d.documentValue}' expected='{d.expectedValue}' origin='{d.actualOrigin}' | {UiText.Deviation(d)}");
        Line($"   evidence count {o.Ui.EvidenceCount}; plan: {plan}");
        if (gatherEvidence)
            Check(o.Ui.EvidenceCount > 0, $"d{day} slot {slot}: a deviation fault ({FaultKey(c)}) is proven before its denial ({o.Ui.EvidenceCount} logged)");

        // A directive fault's rule held against its value on the workbench (the PC workbench spec §4.3), so the PC's Deny can cite it.
        if (!accept && c.HasDirectiveFault)
            yield return CiteRule(o, c, day, slot);

        // Odd slots decide at the stamp tray, even ones on the PC: at the workbench's decision step, where Deny waits for a logged
        // difference or broken rule (an unproven denial, a planned mistake, then goes to the ungated stamp tray).
        int before = o.Game.Ledger.verdicts.Count;
        o.App.Guide.Go(GuideStage.Decision);
        yield return Frames(1);
        Button pcDeny = (Button)Field(o.Ui, "denyButton");
        if (slot % 2 == 0 && !accept)
        {
            Check(pcDeny.interactable == o.App.Board.Log.HasDifference, $"d{day} slot {slot}: the PC's Deny waits for a logged difference ({o.App.Board.Log.Differences} logged, interactable {pcDeny.interactable})");
            if (c.HasDeviationFault && gatherEvidence || c.HasDirectiveFault)
                Check(pcDeny.interactable, $"d{day} slot {slot}: a right denial ({FaultKey(c)}) can be cited on the PC");
        }
        if ((slot % 2 == 1 || (!accept && !pcDeny.interactable)) && o.Tray != null)
        {
            o.Tray.Open();
            yield return Frames(2);
            string opened = o.Tray.IsOpen ? "open" : "NOT OPEN";
            ((Button)Field(o.Tray, accept ? "acceptButton" : "denyButton")).onClick.Invoke();
            Line($"   decided at the stamp tray ({opened}): {(accept ? "ACCEPT" : "DENY")}");
        }
        else
        {
            ((Button)Field(o.Ui, accept ? "acceptButton" : "denyButton")).onClick.Invoke();
            Line($"   decided on the PC: {(accept ? "ACCEPT" : "DENY")}");
        }
        Check(o.Game.Ledger.verdicts.Count == before + 1, $"d{day} slot {slot}: the decision recorded one verdict");
        Decided.Add(c.visitorDisplayName);
        if (!accept && c.legendarySource == null && c.returning == null)
            DeniedFaces[c.caseSeed] = Face(c);
        CaseVerdict v = o.Game.Ledger.verdicts.Last();
        Line($"   verdict: {JsonUtility.ToJson(v)}");
        bool mistake = plan.StartsWith("MISTAKE");
        Check(v.correct != mistake, $"d{day} slot {slot}: the verdict is {(mistake ? "the planned mistake" : "right")} ({plan}; correct={v.correct}, fault '{v.faultReason}', unproven={v.unprovenDenial})");
        var reacted = (DialogRunner)Field(Field(o.Ui, "_interview"), "_runner");
        foreach (DialogLine l in reacted.Transcript.Skip(Math.Max(0, reacted.Transcript.Count - 2)).Where(l => l.Id != null && l.Id.Contains("reactions")))
            Line($"   reaction: [{l.Id}] {l.Text}");
        WorldState w = RunManager.Instance.World;
        Line($"   world: money={w.money} stability={F(w.timelineStability)} citationsToday={w.citationsToday} ending='{w.endingId}'");
        yield return Frames(2);
        if (!string.IsNullOrEmpty(w.endingId))
        {
            Line($"   ENDING {w.endingId}");
            _ended = true;
            yield break;
        }
        if (o.CitationPanel.activeInHierarchy)
        {
            string slip = ((TMP_Text)Field(o.OfficeUi, "citationText")).text;
            Line($"   citation slip: {slip.Replace("\n", " / ")}");
            string rule = v.unprovenDenial ? UiText.Get("citation.rule.evidence") : Citations.RuleLine(c.citation, UiText.Get, UiText.Get("citation.rule.numbered"));
            string value = v.unprovenDenial ? "0" : c.citation?.Values.Select(x => x.Value).FirstOrDefault(x => !string.IsNullOrWhiteSpace(x));
            Check(!string.IsNullOrEmpty(rule) && slip.Contains(rule) && !string.IsNullOrEmpty(value) && slip.Contains(value),
                  $"d{day} slot {slot}: the citation slip names the exact rule ('{rule}') and value ('{value}')");
            if (ShotsOn && First($"citation {v.faultReason}/{v.unprovenDenial}/{v.accepted}"))
                yield return Shots($"citation_{(v.accepted ? "accepted" : "denied")}_{(string.IsNullOrEmpty(v.faultReason) ? "honest" : v.faultReason)}{(v.unprovenDenial ? "_unproven" : "")}");
            ((Button)Field(o.OfficeUi, "citationContinueButton")).onClick.Invoke();
            yield return Frames(2);
        }
        Line($"   verdict line: {((TMP_Text)Field(o.OfficeUi, "resultText")).text.Replace("\n", " / ")}");
    }

    /// <summary>
    /// Wave 5, lesson 3: after the evidence, the wheel's differences menu asks
    /// about each difference the workbench's findings logged as evidence
    /// (Confrontations.About over MatchBoard's FindingLog: one question per
    /// category; the desk names its stated value; the traveller answers, their
    /// outcome counted: the honest explain, a liar cracks or doubles down).
    /// </summary>
    private static IEnumerator Confront(Office o, CaseInstance c, int day, int slot)
    {
        object interview = Field(o.Ui, "_interview");
        var runner = (DialogRunner)Field(interview, "_runner");
        List<Discrepancy> log = o.App.Board.Log.Items.Select(Confrontations.About).Where(x => x != null).ToList();
        var report = (DiscrepancyLog)Field(Field(o.Ui, "_evidence"), "_discrepancies");
        Check(log.Select(x => x.category).Distinct().Count() == report.Count,
              $"d{day} slot {slot}: the findings' evidence ({log.Count}) covers the Deviation Report's {report.Count} categories (one evidence model)");
        // A difference with no prompt of its proof (someone else's photo: who the photo shows is never spoken) raises no question.
        ConfrontWording wording = RunManager.Instance.Library.Interview.confront;
        int askable = log.Where(x => Confrontations.Prompt(wording != null ? wording.prompts : null, x) != null).Select(x => x.category).Distinct().Count();
        if (askable == 0 || !(bool)Field(interview, "_interviewReachable"))
            yield break;
        // Back to the hub first (a garment's look leaves the wheel in the look menu).
        for (int guard = 0; guard < 4 && runner.Choices.All(x => x.Id != InterviewScript.DifferencesNodeId); guard++)
        {
            DialogChoice up = runner.Choices.FirstOrDefault(x => x.Kind == DialogChoiceKind.Back);
            if (up == null)
                break;
            Call(interview, "Choose", up.Id);
        }
        DialogChoice entry = runner.Choices.FirstOrDefault(x => x.Id == InterviewScript.DifferencesNodeId);
        Check(entry != null, $"d{day} slot {slot}: the wheel offers a question about the {log.Count} difference(s) logged as evidence");
        if (entry == null)
            yield break;
        bool shots = ShotsOn && First($"difference menu {c.IsLiar || c.IsForger}");
        if (shots)
        {
            o.Wheel.Open();
            yield return Frames(4);
        }
        Call(interview, "Choose", entry.Id);
        int offered = runner.Choices.Count(x => x.Id.StartsWith("confront:", StringComparison.Ordinal));
        int menuCapacity = RunManager.Instance.Library.Interview.menuCapacity;
        int expected = Math.Min(askable, menuCapacity > 0 ? menuCapacity - 1 : int.MaxValue);
        Check(offered == expected, $"d{day} slot {slot}: the differences menu asks about each category logged as evidence ({offered} offered, {expected} expected)");
        if (shots)
        {
            yield return Frames(4);
            yield return Shots($"d{day}_difference_menu");
        }
        bool cracked = false;
        for (int guard = 0; guard < 12; guard++)
        {
            DialogChoice q = runner.Choices.FirstOrDefault(x => x.Id.StartsWith("confront:", StringComparison.Ordinal));
            if (q == null)
                break;
            Discrepancy d = log.FirstOrDefault(x => Confrontations.ChoiceId(x.category) == q.Id);
            ConfrontOutcome outcome = Confrontations.Outcome(ReactionIntents.Of(c.IsLiar, c.IsForger), c.legendarySource != null, c.confess, c.dialogSeed, d != null ? d.category : default, cracked);
            cracked |= outcome == ConfrontOutcome.Crack;
            int before = runner.Transcript.Count;
            bool shot = ShotsOn && First($"difference {outcome}");
            if (shot && !o.Wheel.IsOpen)
            {
                o.Wheel.Open();
                yield return Frames(4);
            }
            Call(interview, "Choose", q.Id);
            List<DialogLine> said = runner.Transcript.Skip(before).ToList();
            if (shot)
            {
                yield return Seconds(2.5f);
                yield return Shots($"d{day}_difference_{outcome}");
                yield return AppShots(o, c, $"d{day}_difference_{outcome}_transcript", AppTab.Transcript, AppTab.Report);
            }
            Line($"   asked about a difference: '{q.Label}' -> {outcome}: {string.Join(" / ", said.Select(l => $"[{l.Speaker}] {l.Text}"))}");
            Count($"difference answered {outcome}");
            Check(d != null && said.Count >= 2 && said[0].Speaker == DialogSpeaker.Desk && said[0].Text.Contains(d.documentValue) && said.Last().Speaker == DialogSpeaker.Traveller,
                  $"d{day} slot {slot}: the desk names the difference ('{d?.documentValue}') and the traveller answers ({outcome})");
            yield return Seconds(0.3f);
        }
        DialogChoice back = runner.Choices.FirstOrDefault(x => x.Kind == DialogChoiceKind.Back);
        if (back != null)
            Call(interview, "Choose", back.Id);
        Check(runner.Choices.All(x => x.Id != InterviewScript.DifferencesNodeId), $"d{day} slot {slot}: every difference asked, the entry leaves the wheel");
        if (o.Wheel.IsOpen)
            o.Wheel.Close();
        yield return Frames(2);
    }

    private static bool First(string key) => Seen.Add(key);

    /// <summary>The fault a deviation fault carries, for the reports: the lie, a record lie, or the costume error.</summary>
    private static string FaultKey(CaseInstance c) =>
        c.lie.HasValue ? c.lie.Value.ToString() : c.IsForger ? "record lie" : c.costumeFault != CostumeError.None ? "costume " + c.costumeFault : "?";

    /// <summary>Every request at the hub and in the papers menu (a request group as one entry; a form the traveller does not carry gets its missing-form reply), then every question in the ask menu; with <paramref name="dialog"/> also the hub's first dialog, its first choice at each step, to its end.</summary>
    private static IEnumerator Interview(Office o, CaseInstance c, bool dialog)
    {
        var runner = (DialogRunner)Field(Field(o.Ui, "_interview"), "_runner");
        var chosen = new HashSet<string>();
        void Choose(DialogChoice choice)
        {
            chosen.Add(choice.Id);
            Call(Field(o.Ui, "_interview"), "Choose", choice.Id);
        }

        // Every request, in the papers menu ("Request papers >", phase 3) too: out of it by "< Back" once its forms are handed over.
        // The waiver pad's entry is noted, never chosen (report only).
        bool padOffered = false;
        for (int guard = 0; guard < 24; guard++)
        {
            padOffered |= runner.Choices.Any(x => x.Id == InterviewScript.PadChoiceId);
            DialogChoice request = runner.Choices.FirstOrDefault(x => x.Kind == DialogChoiceKind.Request && !chosen.Contains(x.Id) && x.Id != InterviewScript.PadChoiceId);
            if (request == null)
            {
                DialogChoice up = runner.Choices.FirstOrDefault(x => x.Kind == DialogChoiceKind.Back);
                if (up == null)
                    break;
                Call(Field(o.Ui, "_interview"), "Choose", up.Id);
                continue;
            }
            Choose(request);
        }
        Count(padOffered ? "waiver pad offered" : "waiver pad not offered");
        yield return Seconds(0.4f);

        DialogChoice ask = runner.Choices.FirstOrDefault(x => x.Kind == DialogChoiceKind.Question && !string.IsNullOrEmpty(x.Next));
        if (ask != null)
        {
            Choose(ask);
            for (int guard = 0; guard < 16; guard++)
            {
                DialogChoice q = runner.Choices.FirstOrDefault(x => x.Kind == DialogChoiceKind.Question && string.IsNullOrEmpty(x.Next) && !chosen.Contains(x.Id));
                if (q == null)
                    break;
                Choose(q);
            }
            DialogChoice back = runner.Choices.FirstOrDefault(x => x.Kind == DialogChoiceKind.Back);
            if (back != null)
                Choose(back);
        }

        if (dialog)
        {
            DialogChoice start = runner.Choices.FirstOrDefault(x => x.Kind == DialogChoiceKind.Dialog);
            if (start != null)
            {
                Choose(start);
                for (int guard = 0; guard < 16; guard++)
                {
                    DialogChoice next = runner.Choices.FirstOrDefault(x => x.Kind == DialogChoiceKind.Normal);
                    if (next == null)
                        break;
                    Choose(next);
                    if (next.Action == DialogAction.CompleteDialog)
                        break;
                }
            }
        }
        yield return Frames(2);
    }

    /// <summary>The registry's lookup as it read before phase 19 moved it into search (by number first, then by full name, ignoring case): the static dump's FindRecord, so the play finds the same record.</summary>
    private static CitizenRecord FindRecord(CitizenRegistry registry, string query)
    {
        if (registry == null || string.IsNullOrWhiteSpace(query))
            return null;
        string q = query.Trim();
        return registry.Records.FirstOrDefault(r => !string.IsNullOrWhiteSpace(r.Number) && string.Equals(r.Number.Trim(), q, StringComparison.OrdinalIgnoreCase))
               ?? registry.Records.FirstOrDefault(r => string.Equals(r.FullName.Trim(), q, StringComparison.OrdinalIgnoreCase));
    }

    /// <summary>Proves each of a deviation fault's tells through the real compare path: two papers that disagree first (the cross proof, one pair per category), then each tell picked against the claimed place's book row, else the row its value belongs to (or the traveller's own record row for a record category); a dress tell is picked through the wheel's Look menu. A category already documented is not proven twice.</summary>
    private static IEnumerator Evidence(Office o, CaseInstance c, int day, int slot)
    {
        var compare = (CompareController)Field(o.Ui, "compareController");
        var tr = (CaseTranslation)Field(Field(o.Ui, "_interview"), "_caseTranslation");
        var facts = (FactTable)Field(Field(o.Ui, "_reference"), "_facts");
        var runner = (DialogRunner)Field(Field(o.Ui, "_interview"), "_runner");
        var log = (DiscrepancyLog)Field(Field(o.Ui, "_evidence"), "_discrepancies");
        ContentLibrarySO lib = RunManager.Instance.Library;
        bool Documented(ClueCategory category) => log.Items.Any(d => d.category == category);

        ComparePick? Proof(ClueCategory category, string tellValue, bool home)
        {
            if (Forgery.IsRecordCategory(category) || category == ClueCategory.Name)
            {
                CitizenRecord rec = RecordOf(o, c);
                RecordRow? recordRow = rec?.Groups.SelectMany(g => g.Rows).Where(r => r.IsEvidence && r.Category == category).Cast<RecordRow?>().FirstOrDefault();
                return rec != null && recordRow.HasValue ? EvidencePicks.ForRecord(rec, recordRow.Value) : (ComparePick?)null;
            }
            ReferenceBookSO book = lib.ReferenceBooks.FirstOrDefault(b => b != null && b.category == category);
            FactRow row;
            if (home)
            {
                if (!facts.TryFindOtherPlaceWith(category, c.claimedNation.id, c.claimedEra.id, tellValue, out row))
                    return null;
            }
            else
            {
                row = facts.Rows(category).FirstOrDefault(r => r.NationId == c.claimedNation.id && r.EraId == c.claimedEra.id);
                if (row.Value == null)
                    return null;
            }
            return EvidencePicks.ForBookRow(book, row);
        }

        IEnumerator Pair(string what, Func<ComparePick?> tell, ClueCategory category, string tellValue)
        {
            if (Documented(category))
            {
                Line($"   {what}: already documented");
                yield break;
            }
            int before = o.Ui.EvidenceCount;
            foreach (bool home in new[] { false, true })
            {
                ComparePick? proof = Proof(category, tellValue, home);
                if (proof == null)
                    continue;
                compare.Clear();
                ComparePick? first = tell();
                if (first != null)
                    compare.Select(first.Value, null);
                compare.Select(proof.Value, null);
                yield return Frames(1);
                if (o.Ui.EvidenceCount > before)
                    yield break;
            }
            Line($"   evidence FAILED for {what}");
            Check(false, $"d{day} slot {slot}: the {what} tell is proven through the compare path");
        }

        // Two papers that disagree (a borrowed manifest, a doctored ID): the cross proof, one pair per category.
        var done = new HashSet<ClueCategory>();
        foreach (PaperContradiction x in PaperChecks.Contradictions(c.documents.Select(p => (IReadOnlyList<DocumentField>)p.fields).ToList()))
        {
            if (!done.Add(x.Category))
                continue;
            int before = o.Ui.EvidenceCount;
            compare.Clear();
            compare.Select(EvidencePicks.ForField(x.DocumentA, new DocumentRow(x.FieldA, c.documents[x.DocumentA].fields[x.FieldA]), c.documents[x.DocumentA].DisplayName), null);
            compare.Select(EvidencePicks.ForField(x.DocumentB, new DocumentRow(x.FieldB, c.documents[x.DocumentB].fields[x.FieldB]), c.documents[x.DocumentB].DisplayName), null);
            yield return Frames(1);
            if (o.Ui.EvidenceCount == before)
            {
                Line($"   evidence FAILED for the cross proof of {x.Category}");
                Check(false, $"d{day} slot {slot}: the cross proof of {x.Category} logs a deviation");
            }
        }

        // Papers: the first tell field of each category not yet documented (a seal against the Seal Register, a photo against the face).
        for (int d = 0; d < c.documents.Count; d++)
        {
            DocumentInstance doc = c.documents[d];
            for (int f = 0; f < doc.fields.Count; f++)
            {
                if (doc.fields[f] == null)
                    continue;
                var row = new DocumentRow(f, doc.fields[f]); // a paper is one page (the validator): its fields in their order
                if (!row.Field.isAnachronism || !done.Add(row.Field.category))
                    continue;
                int index = d;
                DocumentRow r = row;
                string docName = doc.template != null ? doc.template.displayName : "?";
                if (row.Field.category == ClueCategory.Seal || row.Field.category == ClueCategory.Photo)
                    yield return PairVisual(o, compare, runner, $"papers {row.Field.category} ({docName})", index, r, row.Field.issuer,
                                            row.Field.category == ClueCategory.Seal ? FindingKind.SealIncorrect : FindingKind.Differs, true, day, slot);
                else
                    yield return Pair($"papers {row.Field.category}", () => EvidencePicks.ForField(index, r, docName), row.Field.category, row.Field.value);
            }
        }

        // Answers: the tell line in the transcript (every question was asked).
        foreach (InterviewAnswer a in c.answers.Where(x => x.isTell))
        {
            int i = -1;
            for (int k = 0; k < runner.Transcript.Count; k++)
                if (runner.Transcript[k].IsAnswer && runner.Transcript[k].IsTell && runner.Transcript[k].Category == a.category)
                    i = k;
            if (i < 0)
            {
                Line($"   evidence FAILED: no {a.category} tell line in the transcript");
                Check(false, $"d{day} slot {slot}: the {a.category} answer tell was spoken (its question asked)");
                continue;
            }
            DialogLine line = runner.Transcript[i];
            int index = i;
            yield return Pair($"answer {a.category}", () => EvidencePicks.ForAnswer(index, line, tr), a.category, a.value);
        }

        // Dress: Look > the tell garment (the controller puts it into the compare), then the Costume Guide row.
        if (c.look != null)
        {
            for (int g = 0; g < c.look.Garments.Count; g++)
            {
                Garment garment = c.look.Garments[g];
                if (!garment.IsTell)
                    continue;
                int gi = g;
                ComparePick? LookAt()
                {
                    compare.Clear();
                    DialogChoice look = runner.Choices.FirstOrDefault(x => x.Kind == DialogChoiceKind.Look && !string.IsNullOrEmpty(x.Next));
                    if (look != null)
                        Call(Field(o.Ui, "_interview"), "Choose", look.Id);
                    DialogChoice item = runner.Choices.FirstOrDefault(x => x.Action == DialogAction.InspectGarment && x.GarmentIndex == gi);
                    if (item != null)
                        Call(Field(o.Ui, "_interview"), "Choose", item.Id);
                    else
                        Line($"   no Look choice for garment {gi}");
                    DialogChoice back = runner.Choices.FirstOrDefault(x => x.Kind == DialogChoiceKind.Back);
                    if (back != null && item == null)
                        Call(Field(o.Ui, "_interview"), "Choose", back.Id);
                    return null;
                }
                if (Documented(Looks.EvidenceCategory))
                {
                    Line($"   dress {garment.Slot}: already documented");
                    continue;
                }
                yield return PairDress(o, compare, LookAt, () => Proof(Looks.EvidenceCategory, garment.Value, false), () => Proof(Looks.EvidenceCategory, garment.Value, true), $"dress {garment.Slot}", day, slot);
            }
        }
        foreach (Discrepancy dsc in log.Items)
            Count($"proof {dsc.provedBy}");
    }

    /// <summary>
    /// A seal or a photo held against its truth on the PC workbench (the
    /// document design spec, D4, D8; the PC workbench spec §4), as the clerk
    /// does it: a photo's face first through Look > Their face at the desk
    /// (the wheel holds it), then the paper (scanned) opened from its shelf
    /// chip on the left and, for a seal, the Seal Register from the shelf's
    /// books on the right; the paper's box is clicked in its copy, then the
    /// register's seal of <paramref name="office"/>. The findings must say
    /// <paramref name="expected"/> (with <paramref name="tell"/>: logged as
    /// evidence, the evidence count up), a seal's line between the two boxes
    /// labelled with it. An honest seal is also clicked against another
    /// office's seal: "Another office", nothing logged.
    /// </summary>
    private static IEnumerator PairVisual(Office o, CompareController compare, DialogRunner runner, string what, int document, DocumentRow row, string office,
                                          FindingKind expected, bool tell, int day, int slot)
    {
        bool seal = row.Field.category == ClueCategory.Seal;
        int before = o.Ui.EvidenceCount;
        compare.Clear();
        var documents = (CaseDocumentsPresenter)Field(o.Ui, "_documents");
        if (documents.Papers.State(document) == PaperState.OnDesk)
            Call(documents, "Scan", document);
        if (!seal)
            Face(o, runner);
        yield return OpenApp(o);
        bool opened = PaperBesideRegister(o, document, seal);
        yield return Frames(4);
        string fieldKey = PickKeys.Field(document, row.Index), truthKey = seal ? PickKeys.Seal(office) : PickKeys.Face;
        AppRow box = Row(o, fieldKey), truth = seal ? Row(o, truthKey) : null;
        Check(opened && box != null && box.Pickable && (!seal || (truth != null && truth.Pickable)),
              $"d{day} slot {slot}: {what}: the paper's box{(seal ? " and its office's seal in the Seal Register" : "")} are clickable on the workbench");
        if (box != null)
            box.Pick();
        if (truth != null)
            truth.Pick();
        yield return Frames(2);
        Finding found = o.App.Board.Log.Find(fieldKey, truthKey);
        var lines = (MatchLines)Field(o.App.Board, "lines");
        string want = UiText.Get(FindingRules.LinkKey(expected));
        string label = ((TMP_Text)Field(lines, "labelText")).text;
        string got = found != null ? found.Kind + (found.Proof ? ", evidence" : "") : "nothing logged";
        Report($"   d{day} slot {slot}: {what} on the workbench: {got}, line '{label}' (showing {lines.Showing}), evidence {before} -> {o.Ui.EvidenceCount}");
        Check(found != null && found.Kind == expected && found.Proof == tell, $"d{day} slot {slot}: {what}: the findings log '{want}'{(tell ? " as evidence" : "")} ({got})");
        if (seal)
            Check(lines.Showing && label == want, $"d{day} slot {slot}: {what}: the line joins the two seals labelled '{want}' ('{label}')");
        if (tell && o.Ui.EvidenceCount <= before)
        {
            Line($"   evidence FAILED for {what}");
            Check(false, $"d{day} slot {slot}: the {what} tell is proven through the workbench");
        }
        if (!tell)
            Check(o.Ui.EvidenceCount == before, $"d{day} slot {slot}: {what}: an honest {row.Field.category} proves nothing ({o.Ui.EvidenceCount} evidence)");
        if (ShotsOn && First($"workbench {row.Field.category} {expected}"))
            yield return Shots($"workbench_{row.Field.category}_{expected}");
        if (seal && !tell)
        {
            AgencyOffice other = RunManager.Instance.Library.Agency.offices.FirstOrDefault(x => x != null && x.id != office && x.TryGetSeal(out _));
            AppRow otherSeal = other != null ? Row(o, PickKeys.Seal(other.id)) : null;
            Check(otherSeal != null, $"d{day} slot {slot}: {what}: another office's seal is on the register");
            if (otherSeal != null && box != null)
            {
                int count = o.App.Board.Log.Count;
                string note = UiText.Get(FindingRules.LinkKey(FindingKind.OtherOffice));
                box.Pick();
                otherSeal.Pick();
                yield return Frames(2);
                string drawn = ((TMP_Text)Field(lines, "labelText")).text;
                Check(o.App.Board.Log.Count == count && drawn == note,
                      $"d{day} slot {slot}: {what}: against another office's seal the line reads '{note}' and nothing is logged ('{drawn}', {o.App.Board.Log.Count - count} logged)");
                if (ShotsOn && First("workbench Seal OtherOffice"))
                    yield return Shots("workbench_Seal_OtherOffice");
            }
        }
        compare.Clear();
        yield return CloseApp(o);
    }

    /// From the day each check arrives on the ramp (the photo's, the seal's:
    /// CheckDay), the first accepted traveller whose papers show an honest
    /// photo, and the first with an honest seal, has it held against its truth
    /// on the PC workbench (the document design spec, D4, D8): the photo
    /// against Look > Their face ("Matching data"), the seal against its
    /// office's seal in the Seal Register ("Matching data", then another
    /// office's: "Another office", nothing logged); nothing proved.
    /// </summary>
    private static IEnumerator VisualProbe(Office o, CaseInstance c, int day, int slot)
    {
        var compare = (CompareController)Field(o.Ui, "compareController");
        var runner = (DialogRunner)Field(Field(o.Ui, "_interview"), "_runner");
        foreach ((ClueCategory category, LieKind lie) in new[] { (ClueCategory.Photo, LieKind.SwappedPhoto), (ClueCategory.Seal, LieKind.ForgedSeal) })
        {
            int from = CheckDay(lie);
            if (from == 0 || day < from || Seen.Contains($"visual probe {category}"))
                continue;
            for (int d = 0; d < c.documents.Count; d++)
            {
                int f = c.documents[d].fields.FindIndex(x => x != null && x.category == category && !x.isAnachronism && (category != ClueCategory.Seal || !string.IsNullOrEmpty(x.issuer)));
                if (f < 0)
                    continue;
                First($"visual probe {category}");

                yield return PairVisual(o, compare, runner, $"honest {category} ({c.documents[d].DisplayName})", d, new DocumentRow(f, field), field.issuer, FindingKind.Match, false, day, slot);
                break;
            }
        }
    }

    private static IEnumerator PairDress(Office o, CompareController compare, Func<ComparePick?> look, Func<ComparePick?> claimed, Func<ComparePick?> home, string what, int day, int slot)
    {
        int before = o.Ui.EvidenceCount;
        foreach (Func<ComparePick?> proof in new[] { claimed, home })
        {
            ComparePick? p = proof();
            if (p == null)
                continue;
            look();
            compare.Select(p.Value, null);
            yield return Frames(1);
            if (o.Ui.EvidenceCount > before)
                yield break;
        }
        Line($"   evidence FAILED for {what}");
        Check(false, $"d{day} slot {slot}: the {what} tell is proven through Look and the Costume Guide");
    }

    /// Paper <paramref name="index"/> on the left from its shelf chip and, with
    /// <paramref name="register"/>, the Seal Register on the right from its chip
    /// in the shelf's Books group (checked there from the day the seal check
    /// arrives; before it, checked absent and the right pane left alone);
    /// false when a chip is missing.
    /// </summary>
    private static bool PaperBesideRegister(Office o, int index, bool register)
    {
        var shelf = (ShelfView)Field(o.App, "shelf");
        AppPane l = o.LeftPane, r = o.RightPane;
        ((Button)Field(l, "headerButton")).onClick.Invoke();
        int paper = -1;
        for (int i = 0; i < shelf.Items.Count && paper < 0; i++)
            if (shelf.Items[i].Target.Tab == AppTab.Documents && shelf.Items[i].Target.Item == index)
                paper = i;
        if (paper >= 0)
            shelf.Chips[paper].onClick.Invoke();
        if (!register)
            return paper >= 0;
        ReferenceBookSO seals = RunManager.Instance.Library.ReferenceBooks.FirstOrDefault(b => b != null && b.category == ClueCategory.Seal);
        int book = -1;
        for (int i = 0; seals != null && i < shelf.Items.Count && book < 0; i++)
            if (shelf.Items[i].GroupKey == "shelf.books" && shelf.Items[i].Label == seals.displayName)
                book = i;
        int today = RunManager.Instance.World.day, sealDay = CheckDay(LieKind.ForgedSeal);
        if (sealDay == 0 || today < sealDay)
        {
            Check(book < 0, $"day {today}: the Seal Register is not on the shelf before the seal check arrives (day {sealDay})");
            return paper >= 0;
        }
        Check(book >= 0, $"day {today}: the Seal Register is a document of the shelf's Books group from day {sealDay} ({(seals != null ? seals.displayName : "no Seal book")})");
        if (r == null || !r.gameObject.activeInHierarchy)
            o.App.Run(AppCommand.ToggleSplit);
        if (book >= 0 && r != null)
        {
            ((Button)Field(r, "headerButton")).onClick.Invoke();
            shelf.Chips[book].onClick.Invoke();
        }
        return paper >= 0 && book >= 0;
    }

    /// <summary>
    /// A directive fault cited on the workbench: the first of today's rules
    /// that the traveller breaks (Directives.FaultOf over their facts) held
    /// against the value it is about (MatchBoard.PickRule, then the value
    /// through the compare: a paper's field of the rule's detail, else the
    /// traveller's record row), so the findings log "Breaks the rule".
    /// </summary>
    private static IEnumerator CiteRule(Office o, CaseInstance c, int day, int slot)
    {
        var rules = (IReadOnlyList<TravelRuleSO>)Field(Field(o.Ui, "_reference"), "_rules");
        var compare = (CompareController)Field(o.Ui, "compareController");
        int index = -1;
        for (int i = 0; rules != null && i < rules.Count; i++)
            if (rules[i] != null && rules[i].AppliesTo(c.kind) && (rules[i].IsClosure ? rules[i].Directive.Closes(c.claimedNation != null ? c.claimedNation.id : null, c.claimedEra != null ? c.claimedEra.id : null)
                                                                    : Directives.FaultOf(rules[i].Directive, c.facts) != DirectiveFault.None))
            {
                index = i;
                break;
            }
        if (index < 0)
        {
            Report($"   d{day} slot {slot}: no rule of today breaks for {c.directiveFault}");
            yield break;
        }
        foreach (ClueCategory about in RuleDetails(rules[index].type, c.directiveFault))
            for (int d = 0; d < c.documents.Count; d++)
            {
                List<DocumentField> fields = c.documents[d].fields;
                for (int f = 0; f < fields.Count; f++)
                {
                    if (fields[f] == null || fields[f].category != about)
                        continue;
                    if (rules[index].type == TravelRuleType.PaperDates &&
                        RuleChecks.Check(rules[index].Directive, about, fields[f].value, c.facts, null, null) != RuleVerdict.Breaks)
                        continue;
                    compare.Clear();
                    o.App.Board.PickRule(index, rules[index]);
                    compare.Select(EvidencePicks.ForField(d, new DocumentRow(f, fields[f]), c.documents[d].DisplayName), null);
                    yield return Frames(1);
                    Report($"   d{day} slot {slot}: rule {index + 1} held against {c.documents[d].DisplayName} · {fields[f].label}: {o.App.Board.Log.Differences} difference(s) logged");
                    if (o.App.Board.Log.HasDifference)
                        yield break;
                }
            }
        CitizenRecord rec = RecordOf(o, c);
        if (rec != null)
            foreach (RecordGroup group in rec.Groups)
                foreach (RecordRow row in group.Rows)
                    if (RuleDetails(rules[index].type, c.directiveFault).Contains(row.Category))
                    {
                        compare.Clear();
                        o.App.Board.PickRule(index, rules[index]);
                        compare.Select(EvidencePicks.ForRecord(rec, row), null);
                        yield return Frames(1);
                        Report($"   d{day} slot {slot}: rule {index + 1} held against the record's {row.Label}: {o.App.Board.Log.Differences} difference(s) logged");
                        if (o.App.Board.Log.HasDifference)
                            yield break;
                    }
    }

    /// <summary>The details a rule of <paramref name="type"/> is about (RuleChecks), the fault's own first.</summary>
    private static List<ClueCategory> RuleDetails(TravelRuleType type, DirectiveFault fault)
    {
        switch (type)
        {
            case TravelRuleType.PaperDates:
                return fault == DirectiveFault.ExpiredPaper ? new List<ClueCategory> { ClueCategory.Expiry, ClueCategory.DepartureDate }
                                                            : new List<ClueCategory> { ClueCategory.DepartureDate, ClueCategory.Expiry };
            case TravelRuleType.PaperSet:
                return new List<ClueCategory> { ClueCategory.AccountStatus, ClueCategory.TransponderClass, ClueCategory.Signature, ClueCategory.WaiverNo };
            case TravelRuleType.DebtStanding:
                return new List<ClueCategory> { ClueCategory.AccountStatus, ClueCategory.Debt };
            case TravelRuleType.TransponderRecall:
                return new List<ClueCategory> { ClueCategory.TransponderId };
            default:
                return new List<ClueCategory> { ClueCategory.Destination };
        }
    }

    private static IEnumerator Shot(string file)
    {
        string path = Path.Combine(ShotsDir, file);
        if (File.Exists(path))
            File.Delete(path);
        yield return Frames(2);
        float compiling = Time.realtimeSinceStartup + 60f;
        while (ShaderUtil.anythingCompiling && Time.realtimeSinceStartup < compiling)
            yield return null;
        yield return Frames(2);
        ScreenCapture.CaptureScreenshot(path);
        float end = Time.realtimeSinceStartup + 10f;
        while (!File.Exists(path) && Time.realtimeSinceStartup < end)
            yield return null;
        Report($"screenshot {file} (screen {UnityEngine.Screen.width}x{UnityEngine.Screen.height}){(File.Exists(path) ? "" : " NOT WRITTEN")}");
        if (_frameOpen)
            ShotDesktop("desktop_" + file);
    }

    /// <summary>The PC desktop as the frame's glass should show it: the desktop camera rendered to a texture the screen's height tall at the desktop's 4:3 aspect, saved beside the screenshot (the screen capture of the open frame is kept too, so what the glass really shows stays on record).</summary>
    private static void ShotDesktop(string file)
    {
        int mask = 1 << OfficeLayers.PcDesktopLayer;
        Camera cam = Object.FindObjectsByType<Camera>(FindObjectsInactive.Include, FindObjectsSortMode.None)
            .Where(c => (c.cullingMask & mask) != 0).OrderBy(c => c.targetTexture != null ? 1 : 0).FirstOrDefault();
        if (cam == null)
        {
            Report($"   no desktop camera for {file}");
            return;
        }
        int height = UnityEngine.Screen.height, width = Mathf.RoundToInt(height * 4f / 3f);
        RenderTexture rt = RenderTexture.GetTemporary(width, height, 24);
        RenderTexture old = cam.targetTexture;
        Rect rect = cam.rect;
        bool wasEnabled = cam.enabled;
        cam.targetTexture = rt;
        cam.rect = new Rect(0f, 0f, 1f, 1f);
        cam.enabled = true;
        cam.Render();
        cam.targetTexture = old;
        cam.rect = rect;
        cam.enabled = wasEnabled;
        RenderTexture active = RenderTexture.active;
        RenderTexture.active = rt;
        var tex = new Texture2D(width, height, TextureFormat.RGB24, false);
        tex.ReadPixels(new Rect(0, 0, width, height), 0, 0);
        tex.Apply();
        RenderTexture.active = active;
        File.WriteAllBytes(Path.Combine(ShotsDir, file), tex.EncodeToPNG());
        Object.Destroy(tex);
        RenderTexture.ReleaseTemporary(rt);
        Report($"desktop {file} ({width}x{height}, camera {cam.name})");
    }

    /// <summary>The House (Home's upgrade tree): every Home upgrade's state and price; the first one buyable with 60 cr to spare is bought (one a night), then on to the slot step.</summary>
    private static IEnumerator House(HomeUIController ui, HomeManager manager, int day)
    {
        var panel = (GameObject)Field(ui, "shopPanel");
        if (!panel.activeInHierarchy)
        {
            Line("   house: not shown");
            yield break;
        }
        WorldState w = RunManager.Instance.World;
        ContentLibrarySO lib = RunManager.Instance.Library;
        List<UpgradeSO> home = lib.Upgrades.Where(u => u != null && u.venue == UpgradeVenue.Home).ToList();
        Line($"   house: {string.Join(", ", home.Select(u => $"{u.id}={OrderBook.StateOf(w, lib, u)}/{OrderBook.Price(w, lib, u)}"))}");

            yield return Shots($"house_night{day}");

            yield return Frames(2);
            if (ShotsOn && First("house bought"))
                yield return Shots($"house_bought_night{day}");
        }
        ((Button)Field(ui, "shopContinueButton")).onClick.Invoke();
        yield return Frames(3);
    }

    /// <summary>The world's answers latched last night (dated today) print their headlines in this morning's paper, within the history news cap.</summary>
    private static void WorldHeadlines(int day)
    {
        WorldState w = RunManager.Instance.World;
        WorldContent content = RunManager.Instance.Library.World;
        List<FactorLead> today = w.leads.Where(l => l.sinceDay == day).ToList();
        if (today.Count == 0)
            return;
        string paper = ((TMP_Text)Field(Office.One<DayFlowUIController>(), "briefingBodyText")).text;
        // Wave 5, lesson 10: each headline ends with the last traveller whose verdict pulled it, and the day.
        string traced = RunManager.Instance.Library.HistoryLines.traced?.text;
        List<string> headlines = today.Select(l => Traces.Traced(content.Headline(l), traced, Traces.Of(w.pullTraces, l))).Where(h => !string.IsNullOrWhiteSpace(h)).ToList();
        int printed = headlines.Count(h => paper.Contains(h));
        int named = today.Count(l => Traces.Of(w.pullTraces, l) != null);
        Line($"   world headlines today: {printed} of {headlines.Count} printed, {named} traced to a traveller");
        if (named > 0)
            Count("world headline traced to a traveller");
        Count("world headline");
        Check(printed > 0 || headlines.Count == 0, $"day {day}: a world answer changed overnight and the paper prints its headline ({printed} of {headlines.Count})");
    }

    private static void RecordSave(string name)
    {
        if (!File.Exists(SaveSystem.SavePath))
        {
            Line($"   save {name}: NO SAVE FILE");
            return;
        }
        string json = File.ReadAllText(SaveSystem.SavePath);
        File.WriteAllText(Path.Combine(Dir, "saves", name + ".json"), json);
        using var sha = SHA1.Create();
        Line($"   save {name}: {json.Length} chars sha1 {BitConverter.ToString(sha.ComputeHash(Encoding.UTF8.GetBytes(json))).Replace("-", "").ToLowerInvariant()}");
    }

    private static IEnumerable<string> Lines(string text) =>
        (text ?? string.Empty).Replace("\r", "").Split('\n').Where(l => l.Length > 0);

    private static string F(float f) => f.ToString("0.###", System.Globalization.CultureInfo.InvariantCulture);

    private static string Fields(object o)
    {
        if (o == null)
            return "null";
        return string.Join(" ", o.GetType().GetFields(BindingFlags.Instance | BindingFlags.Public | BindingFlags.NonPublic)
            .Select(f => $"{f.Name}={Value(f.GetValue(o))}"));
    }

    private static string Value(object v) => v switch
    {
        null => "null",
        float f => F(f),
        string s => $"'{s}'",
        System.Collections.IEnumerable e => "[" + string.Join(",", e.Cast<object>().Select(x => x is float xf ? F(xf) : x?.ToString())) + "]",
        _ => v.ToString()
    };

    private static IEnumerator Frames(int n)
    {
        int target = Time.frameCount + n;
        while (Time.frameCount < target)
            yield return null;
    }

    private static IEnumerator Seconds(float seconds)
    {
        float end = Time.realtimeSinceStartup + seconds;
        while (Time.realtimeSinceStartup < end)
            yield return null;
    }

    private static IEnumerator Until(Func<bool> condition, float timeout, string what)
    {
        float end = Time.realtimeSinceStartup + timeout;
        while (!condition())
        {
            if (Time.realtimeSinceStartup > end)
                throw new TimeoutException($"waited {timeout} s for {what}");
            yield return null;
        }
    }

    private static void Finish()
    {
        PlayModeWindow.SetCustomRenderingResolution((uint)SessionState.GetInt(ViewWidthKey, 1920), (uint)SessionState.GetInt(ViewHeightKey, 1080), ViewName);
        int hadSave = SessionState.GetInt(HadSaveKey, 0);
        for (int i = 0; i < SaveSystem.Files.Count; i++)
        {
            bool had = (hadSave & (1 << i)) != 0;
            if (had && File.Exists(SaveBackup + i))
            {
                File.Copy(SaveBackup + i, SaveSystem.Files[i], true);
                File.Delete(SaveBackup + i);
            }
            else if (!had && File.Exists(SaveSystem.Files[i]))
            {
                File.Delete(SaveSystem.Files[i]);
            }
        }
        foreach ((string key, string pref) in new[] { (PrefKey, UiLanguagePref), (MotionKey, MotionPref) })
        {
            string value = SessionState.GetString(key, "<absent>");
            if (value == "<absent>")
                PlayerPrefs.DeleteKey(pref);
            else
                PlayerPrefs.SetString(pref, value);
        }
        foreach (string entry in SessionState.GetString(DesktopPrefsKey, string.Empty).Split('\u0002').Where(e => e.Length > 0))
        {
            string[] kv = entry.Split('\u0001');
            if (kv.Length != 2)
                continue;
            if (kv[1] == "<absent>")
                PlayerPrefs.DeleteKey(kv[0]);
            else
                PlayerPrefs.SetString(kv[0], kv[1]);
        }
        PlayerPrefs.Save();
        Resources.Load<RunConfigSO>(RunManager.ConfigResourcePath).fixedRunSeed = SessionState.GetInt(SeedKey, 0);

        // Errors and exceptions logged during the play are failures (the warnings file keeps them all).
        string[] logged = File.Exists(WarningsPath) ? File.ReadAllLines(WarningsPath) : new string[0];
        int errors = logged.Count(l => l.StartsWith("[Error]") || l.StartsWith("[Exception]") || l.StartsWith("[Assert]"));
        Check(errors == 0, $"the log holds no error or exception ({errors} of {logged.Length} lines)");

        Report($"fails={SessionState.GetInt(FailsKey, 0)}");
        SessionState.SetInt(StepKey, 0);
        EditorApplication.update -= Tick;
        Application.logMessageReceived -= OnLog;
        string scene = SessionState.GetString(SceneKey, string.Empty);
        if (!string.IsNullOrEmpty(scene) && File.Exists(scene))
            EditorSceneManager.OpenScene(scene, OpenSceneMode.Single);
        else
            EditorSceneManager.NewScene(NewSceneSetup.EmptyScene, NewSceneMode.Single);
        Report($"done {DateTime.Now:s}");
    }

    private static void OnLog(string message, string stack, LogType type)
    {
        if (type == LogType.Log)
            return;
        if (stack != null && stack.Contains("UnityEditor.Search"))
            return;
        if (message.StartsWith("Account API did not become accessible"))
            return;
        try { File.AppendAllText(WarningsPath, $"[{type}] {message}\n"); } catch (IOException) { }
    }

    private static object Field(object target, string name)
    {
        for (Type t = target.GetType(); t != null; t = t.BaseType)
        {
            FieldInfo f = t.GetField(name, BindingFlags.Public | BindingFlags.NonPublic | BindingFlags.Instance);
            if (f != null)
                return f.GetValue(target);
        }
        throw new MissingFieldException(target.GetType().Name, name);
    }

    private static object Call(object target, string name, params object[] args)
    {
        MethodInfo m = target.GetType().GetMethods(BindingFlags.Public | BindingFlags.NonPublic | BindingFlags.Instance)
            .First(x => x.Name == name && x.GetParameters().Length == args.Length);
        return m.Invoke(target, args);
    }

    private static void Line(string line) => File.AppendAllText(TranscriptPath, line + "\n");

    private static void Report(string line) => File.AppendAllText(ReportPath, line + "\n");

    private static void Check(bool ok, string what)
    {
        Report((ok ? "PASS " : "FAIL ") + what);
        if (!ok)
            SessionState.SetInt(FailsKey, SessionState.GetInt(FailsKey, 0) + 1);
    }
